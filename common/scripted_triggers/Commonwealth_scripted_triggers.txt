# State scope; ROOT is the country. A core state with room for another generator.
COM_is_autarky_supply_site = {
	is_owned_by = ROOT
	is_controlled_by = ROOT
	is_core_of = ROOT
	OR = {
		free_building_slots = { building = water_generator size > 0 }
		free_building_slots = { building = electricity_generator size > 0 }
		free_building_slots = { building = metal_generator size > 0 }
		free_building_slots = { building = composites_generator size > 0 }
		free_building_slots = { building = circuitry_generator size > 0 }
	}
}
