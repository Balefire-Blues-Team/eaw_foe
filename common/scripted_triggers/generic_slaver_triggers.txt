# Country scope: the market is open to us
GSL_market_open = {
	has_country_flag = slavery_allowed
	# Rustwreck lends its labourers out on terms and does not trade in slaves
	NOT = { original_tag = RWK }
	OR = {
		NOT = { has_focus_tree = GSL_focus }
		has_completed_focus = GSL_the_auction_block
	}
}

# Country scope
GSL_can_buy_factories = {
	check_variable = { global.GSL_pool > 0.99 }
	num_of_available_civilian_factories > 0
	NOT = { check_variable = { GSL_debt_days > 360 } }
}

# Country scope
GSL_can_buy_pp = {
	check_variable = { global.GSL_pool > 0.99 }
	check_variable = { has_political_power > global.GSL_buy_pp }
}

# Country scope
GSL_can_buy_guns = {
	check_variable = { global.GSL_pool > 0.99 }
	check_variable = { num_equipment@infantry_equipment > global.GSL_buy_guns }
}
