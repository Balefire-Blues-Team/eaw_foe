# All these triggers run in country scope.

# STATE scope. True if the state has an owner. Many wasteland states have none, and an owner or controller
# scope on them logs "invalid event target"; check this first.
FOE_state_has_owner = {
	NOT = { check_variable = { owner = 0 } }
}

# COUNTRY scope. True if the country has a leader character. Some countries have none, and a
# var:country_leader scope on them logs "Invalid Scope, supported: Character, provided: Country"; check this first.
FOE_has_leader_character = {
	NOT = { check_variable = { country_leader = 0 } }
}

# Class tiers. A combo class is always Mythic, and counts as Mythic in both of its parent classes:
# Ruler = Conqueror + Administrator, Exploiter = Conqueror + Industrialist, Magnate = Administrator + Industrialist.

		has_country_leader_with_trait = great_leader_military_trait
		has_country_leader_with_trait = legendary_leader_military_trait

		has_completed_focus = GR_basecamp
		has_completed_focus = gs_opening1
		has_completed_focus = tribal_past
		has_completed_focus = genstable_opening1
	}
	has_war = no
	is_in_faction = no
	is_subject = no
	any_owned_state = {
		any_neighbor_state = {

		}
	}
	custom_trigger_tooltip = {
		tooltip = FOE_unifier_absorb_size_tt
		set_temp_variable = { FOE_absorb_state_limit = ROOT.num_core_states }
		divide_temp_variable = { FOE_absorb_state_limit = 2 }
		check_variable = { var = num_owned_states value = FOE_absorb_state_limit compare = less_than_or_equals }
	}
}

# Country scope; ROOT is the Puppetmaster.
FOE_is_puppet_install_target = {
	exists = yes
	is_ai = yes
	# Only generic nations; countries with unique content are never installed.
	OR = {
		has_completed_focus = GR_basecamp
		has_completed_focus = gs_opening1
		has_completed_focus = tribal_past
		has_completed_focus = genstable_opening1
	}
	has_war = no
	is_in_faction = no
	is_subject = no
	# Borders us or one of our subjects.
	any_owned_state = {
		any_neighbor_state = {
			OR = {
				is_owned_by = ROOT
				AND = {
					FOE_state_has_owner = yes
					owner = { is_subject_of = ROOT }
				}
			}
		}
	}
	custom_trigger_tooltip = {
		tooltip = FOE_puppet_install_size_tt
		set_temp_variable = { FOE_install_state_limit = ROOT.num_owned_states }
		add_to_temp_variable = { FOE_install_state_limit = ROOT.num_subjects }
		divide_temp_variable = { FOE_install_state_limit = 2 }
		check_variable = { var = num_owned_states value = FOE_install_state_limit compare = less_than_or_equals }
	}
}

# State scope; the capital. Each category step needs more residents.
FOE_capital_can_rise = {
	OR = {
		AND = {
			NOT = {
				has_state_category = shanty
				has_state_category = town
				has_state_category = large_town
				has_state_category = city
				has_state_category = large_city
				has_state_category = metropolis
				has_state_category = megalopolis
				has_state_category = prewar_megacity
			}
			state_population > 1999
		}
		AND = {
			has_state_category = shanty
			state_population > 3999
		}
		AND = {
			has_state_category = town
			state_population > 5999
		}
		AND = {
			has_state_category = large_town
			state_population > 7999
		}
		AND = {
			has_state_category = city
			state_population > 10999
		}
		AND = {
			has_state_category = large_city
			state_population > 13999
		}
		AND = {
			has_state_category = metropolis
			state_population > 17999
		}
	}
}

#Co-op subclasses: shared safeguards. Every co-op subclass (a Great Leader that elevates or binds another nation)

# Lord of the Long March
# Country scope unless explicitly noted. ROOT must remain the marching country.
FOE_is_lord_of_the_long_march = {
	FOE_has_leader_character = yes
	var:country_leader = {
		has_character_flag = FOE_great_leader_character
		has_character_flag = FOE_lord_of_the_long_march_character
	}
}
# Probe uses a temporary variable only, as in live coring scripted triggers. Mirrors FOE_plan_long_march: a
# three-state route may start next to any state we own, and a different peaceful neighbour must exist as custodian.
FOE_march_probe_route_state = {
	impassable = no
	NOT = { is_owned_by = ROOT }
	FOE_state_has_owner = yes
	owner = {
		FOE_march_hostile_candidate = yes
		has_capitulated = no
	}
}
FOE_can_assign_long_march = {
	FOE_great_leader_country_valid = yes
	is_subject = no
	is_in_faction = no
	has_capitulated = no
	has_war = no
	num_divisions > 0
	NOT = { has_country_flag = FOE_march_committed }
	any_owned_state = {
		any_neighbor_state = {
			FOE_march_probe_route_state = yes
			ROOT = { set_temp_variable = { FOE_march_probe_first = PREV.id } }
			any_neighbor_state = {
				FOE_march_probe_route_state = yes
				any_neighbor_state = {
					FOE_march_probe_route_state = yes
					NOT = { state = var:ROOT.FOE_march_probe_first }
				}
			}
		}
	}
	any_neighbor_country = { FOE_march_custodian_basic = yes }
}
FOE_assignment_suits_long_march = {
	OR = { has_idea = raider_society has_idea = tribal_society check_variable = { num_owned_states < 6 } }
}
FOE_march_can_administer = {
	FOE_is_lord_of_the_long_march = yes

	has_capitulated = no
	has_war = no
}

# Foreign COUNTRY scope. Owns no part of the route; also used to replace a lost custodian.
FOE_march_custodian_candidate = {
	FOE_march_custodian_basic = yes
	NOT = { is_in_array = { array = ROOT.FOE_march_route_owners value = THIS.id } }
	NOT = { var:ROOT.FOE_march_route_1 = { is_owned_by = PREV } }
	NOT = { var:ROOT.FOE_march_route_2 = { is_owned_by = PREV } }
	NOT = { var:ROOT.FOE_march_route_3 = { is_owned_by = PREV } }
}
# The custodian must still border our realm. It need not border the current camp: a small capital may have
# only one foreign neighbour, and that one is often the first waypoint.

	}
