# All these triggers run in country scope.
FOE_is_nomadic_khan = {
	var:country_leader = {
		has_character_flag = FOE_great_leader_character
		has_character_flag = FOE_nomadic_khan_character
	}
}

FOE_is_great_nomadic_khan = {
	FOE_is_nomadic_khan = yes
	has_country_leader_with_trait = great_leader_military_trait
	NOT = { has_country_leader_with_trait = legendary_leader_military_trait }
	NOT = { has_country_leader_with_trait = mythic_leader_military_trait }
}

FOE_is_legendary_nomadic_khan = {
	FOE_is_nomadic_khan = yes
	has_country_leader_with_trait = legendary_leader_military_trait
	NOT = { has_country_leader_with_trait = mythic_leader_military_trait }
}

FOE_is_mythic_nomadic_khan = {
	FOE_is_nomadic_khan = yes
	has_country_leader_with_trait = mythic_leader_military_trait
}

# PREV is the country invoking this trigger, even inside an on-action.
FOE_nomadic_court_controlled = {
	has_variable = FOE_nomadic_court_state
	var:FOE_nomadic_court_state = { is_controlled_by = PREV }
}
