# All these triggers run in country scope.
FOE_is_nomadic_khan = {
	var:country_leader = {
		has_character_flag = FOE_great_leader_character
		has_character_flag = FOE_nomadic_khan_character
	}
}

FOE_is_great_nomadic_khan = {
	FOE_is_nomadic_khan = yes
	has_country_leader_with_trait = great_leader_military_trait
	NOT = { has_country_leader_with_trait = legendary_leader_military_trait }
	NOT = { has_country_leader_with_trait = mythic_leader_military_trait }
}

}

# Country scope; ROOT is the Unifier.
FOE_is_unifier_absorption_target = {
	is_ai = yes
	# Only generic nations; countries with unique content are never absorbed.
	OR = {
		has_completed_focus = GR_basecamp
		has_completed_focus = gs_opening1
		has_completed_focus = tribal_past
		has_completed_focus = genstable_opening1
	}
	has_war = no
	is_in_faction = no
	is_subject = no
	any_owned_state = {
		any_neighbor_state = {
			is_owned_by = ROOT
			is_core_of = ROOT
		}
	}
	custom_trigger_tooltip = {
		tooltip = FOE_unifier_absorb_size_tt
		set_temp_variable = { FOE_absorb_state_limit = ROOT.num_core_states }
		divide_temp_variable = { FOE_absorb_state_limit = 2 }
		check_variable = { var = num_owned_states value = FOE_absorb_state_limit compare = less_than_or_equals }
	}
}

# State scope; ROOT is the Puppetmaster. The state must be a core of a country that no longer exists.
FOE_is_puppet_release_target = {
	is_owned_by = ROOT
	is_controlled_by = ROOT
	NOT = { is_core_of = ROOT }
	any_of_scopes = {
		array = core_countries
		exists = no
	}
}

# State scope.
FOE_has_free_factory_slot = {
	free_building_slots = {
		building = industrial_complex
		size > 0
	}
}

# Country scope; ROOT is the Puppetmaster.
FOE_is_puppet_install_target = {
	exists = yes
	is_ai = yes
	# Only generic nations; countries with unique content are never installed.
	OR = {
		has_completed_focus = GR_basecamp
		has_completed_focus = gs_opening1
		has_completed_focus = tribal_past
		has_completed_focus = genstable_opening1
	}
	has_war = no
	is_in_faction = no
	is_subject = no
	# Borders us or one of our subjects.
	any_owned_state = {
		any_neighbor_state = {
			OR = {
				is_owned_by = ROOT
				owner = { is_subject_of = ROOT }
			}
		}
	}
	custom_trigger_tooltip = {
		tooltip = FOE_puppet_install_size_tt
		set_temp_variable = { FOE_install_state_limit = ROOT.num_owned_states }
		add_to_temp_variable = { FOE_install_state_limit = ROOT.num_subjects }
		divide_temp_variable = { FOE_install_state_limit = 2 }
		check_variable = { var = num_owned_states value = FOE_install_state_limit compare = less_than_or_equals }
	}
}

# State scope; ROOT is the Architect. The capital needs a free factory slot.
FOE_is_industry_relocation_source = {
	is_owned_by = ROOT
	is_controlled_by = ROOT
	is_capital = no
	OR = {
		check_variable = { industrial_complex_level > 0 }
		check_variable = { arms_factory_level > 0 }
	}
	ROOT = {
		capital_scope = { FOE_has_free_factory_slot = yes }
	}
}

# State scope; the capital. Each category step needs more residents.
FOE_capital_can_rise = {
	OR = {
		AND = {
			NOT = {
				has_state_category = shanty
				has_state_category = town
				has_state_category = large_town
				has_state_category = city
				has_state_category = large_city
				has_state_category = metropolis
				has_state_category = megalopolis
				has_state_category = prewar_megacity
			}
			state_population > 1999
		}
		AND = {
			has_state_category = shanty
			state_population > 3999
		}
		AND = {
			has_state_category = town
			state_population > 5999
		}
		AND = {
			has_state_category = large_town
			state_population > 7999
		}
		AND = {
			has_state_category = city
			state_population > 10999
		}
		AND = {
			has_state_category = large_city
			state_population > 13999
		}
		AND = {
			has_state_category = metropolis
			state_population > 17999
		}
	}
}
