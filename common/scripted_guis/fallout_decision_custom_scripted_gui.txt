
scripted_gui = {
	fallout_inventory_window = {
		context_type = player_context
		parent_window_name = "event_header"
		window_name = "peace_quotes_window"

		visible = {
			always = yes
		}

		effects = {
			peace_quotes_checkbox_click = {
				if = { limit = { check_variable = { peace_message_var = 1 } }
					set_variable = { peace_message_var = 2 }
				}
				else_if = { limit = { check_variable = { peace_message_var = 2 } }
					set_variable = { peace_message_var = 3 }
				}
				else = {
					set_variable = { peace_message_var = 1 }
				}
			}
		}
		properties = {
			peace_quotes_checkbox = {
				frame = peace_message_var
			}
		}
	}
}
