		window_name = "sez_city_construction_menu_window"

		}

		effects = {
			close_button_click = {
				SEZ_clear_district_selection = yes
			}
			core_button_click = {
				hidden_effect = {
					add_political_power = -20
					every_owned_state = {
						limit = { has_state_flag = SEZ_state_inspected_flag }
						add_core_of = ROOT
						SEZ_refresh_district_readout = yes
					}
				}
			}
			upgrade_civi_button_click = {
				hidden_effect = {
					add_to_variable = { SEZ_total_upgrades = 1 }
					every_owned_state = {
						limit = { has_state_flag = SEZ_state_inspected_flag }
						add_to_variable = { SEZ_civilian_dm = 0.1 }
						add_building_construction = {
							type = infrastructure
							level = 1
						}
						SEZ_spend_upgrade_point = yes
					}
				}
			}
			upgrade_defense_button_click = {
				hidden_effect = {
					add_to_variable = { SEZ_total_upgrades = 1 }
