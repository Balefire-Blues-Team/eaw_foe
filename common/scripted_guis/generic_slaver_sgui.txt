### Slave market panel, shown at the top of the Slave Trade decision category ###
scripted_gui = {

	GSL_slave_market_ui = {
		context_type = decision_category
		window_name = "GSL_slave_market_window"

		visible = {
			GSL_market_open = yes
		}

		effects = {
			GSL_autorenew_button_click = {
				if = {
					limit = { has_country_flag = GSL_autorenew_flag }
					clr_country_flag = GSL_autorenew_flag
					clr_country_flag = GSL_autorenew_paused_flag
				}
				else = {
					set_country_flag = GSL_autorenew_flag
				}
			}
			GSL_sell_surplus_button_click = {
				GSL_trade_sell_surplus = yes
			}
			GSL_sell_pp_button_click = {
				GSL_trade_sell_pp = yes
			}
			GSL_sell_guns_button_click = {
				GSL_trade_sell_guns = yes
			}
			GSL_buy_factories_button_click = {
				GSL_trade_buy_factories = yes
			}
			GSL_buy_pp_button_click = {
				GSL_trade_buy_pp = yes
			}
			GSL_buy_guns_button_click = {
				GSL_trade_buy_guns = yes
			}
		}

		triggers = {
			# Red Eye does not sell: the sell buttons give way to a line saying so
			GSL_sell_surplus_button_visible = {
				NOT = { original_tag = EYE }
			}
			GSL_sell_pp_button_visible = {
				NOT = { original_tag = EYE }
			}
			GSL_sell_guns_button_visible = {
				NOT = { original_tag = EYE }
			}
			GSL_market_no_sell_visible = {
				original_tag = EYE
			}
			GSL_sell_surplus_button_click_enabled = {
				GSL_can_sell_surplus = yes
			}
			GSL_sell_pp_button_click_enabled = {
				GSL_can_sell_pp = yes
			}
			GSL_sell_guns_button_click_enabled = {
				GSL_can_sell_guns = yes
			}
			GSL_buy_factories_button_click_enabled = {
				GSL_can_buy_factories = yes
			}
			GSL_buy_pp_button_click_enabled = {
				GSL_can_buy_pp = yes
			}
			GSL_buy_guns_button_click_enabled = {
				GSL_can_buy_guns = yes
			}
		}
	}
