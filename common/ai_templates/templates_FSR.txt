# FSR power armour, modelled on the Sheep Mod's Poland templates (2026-09-28).
# Width 20 (10 power armour) won the calculator's width round robin; PA fireteam / demolition regiments lost at
# every width, so only their support companies are used. Strategies: ai_strategy/FSR_army.txt.
# Fifth support company: unicorn_company, th branch. Against EYE it ties with dogs once researched
# (scenarios/fsr_width/run_vs_eye_tech.py) but needs no extra production line and its techs are dated 1235.
#
# Stages: the upgrade_prio tech gates decide which target new templates aim for. Fielded divisions only move to a
# new template through replace_with + can_upgrade_in_field on the template they match (vanilla ai_templates
# _documentation.md; the AI copies templates, never edits them). The starting "PA Company" best matches
# FSR_power_armour_early (imgui show ai_templates, 2026-09-28), so it upgrades in the field through that entry too.
power_armour_FSR = {

	available_for = {
		FSR
	}

	role = power_armour

	upgrade_prio = {
		factor = 2000
	}

	# What FSR can field at the start
	FSR_power_armour_early = {

		upgrade_prio = {
			factor = 3000

			modifier = {
				factor = 0
				has_tech = fireteam_weaponry_tech
				has_tech = demolitions_weaponry_tech
				has_tech = anti_tank_equipment_tech_1
				has_tech = tech_unicorn
			}
		}

		# field upgrade to the full template once researched and the support equipment is in stock
		can_upgrade_in_field = {
			has_tech = fireteam_weaponry_tech
			has_tech = demolitions_weaponry_tech
			has_tech = anti_tank_equipment_tech_1
			has_tech = tech_unicorn
			has_equipment = { fireteam_equipment > 99 }
			has_equipment = { demolitions_equipment > 99 }
			has_equipment = { anti_tank_equipment > 49 }
		}

		target_template = {

			support = {
				power_armour_support = 1
			}

			regiments = {
				power_armour = 10
			}
		}

		replace_at_match = 0.95
		replace_with = FSR_power_armour_full
		target_min_match = 0.9
	}

	# The same division with fireteam, demolition, anti-tank and unicorn support
	FSR_power_armour_full = {

		upgrade_prio = {
			factor = 1000

			modifier = {
				factor = 0
				NOT = {
					has_tech = fireteam_weaponry_tech
					has_tech = demolitions_weaponry_tech
					has_tech = anti_tank_equipment_tech_1
					has_tech = tech_unicorn
				}
			}
		}

		target_template = {

			support = {
				power_armour_support = 1
				fireteam_support = 1
				demolitions_support = 1
				anti_tank_company = 1
				unicorn_company = 1
			}

			regiments = {
				power_armour = 10
			}
		}
	}
}
