robot_combat_default_template = {

	blocked_for = {
		FSR # own templates in templates_FSR.txt
		SEZ # own templates in templates_SEZ.txt
	}

	role = combat_robot

	upgrade_prio = {
		factor = 2.17

		modifier = {
			factor = 7
				has_tech = robot_protectapony_tech
		}
	}

	combat_robot_default = {
		reinforce_prio = 1
		custom_icon = 4

		upgrade_prio = {
			factor = 2

			modifier = {
				factor = 4
			}
		}

		can_upgrade_in_field = {
			has_tech = robot_industrial_robots_tech
			has_equipment = { heavy_robot_equipment > 199 }
		}

		target_template = {
			
			support = {
				support_robot_def = 1
				support_robot_off = 1
				anti_tank_company = 1
				dog_company = 1
			}

			regiments = {
				medium_robot = 8
			}
		}

		replace_at_match = 0.95
		replace_with = combat_robot_heavy
		target_min_match = 0.9
	}
	
	combat_robot_heavy = {
		reinforce_prio = 1
		custom_icon = 4

		upgrade_prio = {
			factor = 2

			modifier = {
				factor = 40
				has_tech = robot_industrial_robots_tech
			}

			modifier = {
				factor = 0
				NOT = { has_tech = robot_industrial_robots_tech }
			}
		}

		# no replace_with: last stage, so this field-upgrade check has no effect
		can_upgrade_in_field = {
			has_equipment = { heavy_robot_equipment > 199 }
		}

		target_template = {
			support = {
				support_robot_def = 1
				support_robot_off = 1
				anti_tank_company = 1
				dog_company = 1
			}

			regiments = {
				medium_robot = 4
				heavy_robot = 10
			}
		}

	}
}
