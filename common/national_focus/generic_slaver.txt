### Generic slaver tree (2026-10-14), built from shared foci so slaver nations can get auxiliary trees.
### Roots: GSL_where_the_chains_came_from (history, opening, supply, estate, three directions, endgame),
### GSL_who_holds_the_whip (politics, opens after the third history focus), GSL_guns_for_the_overseers (army).
### Mechanics (gangs, The Pens, slave market, expeditions) live in generic_slaver_effects.txt and foe_slavery_decisions.txt.

focus_tree = {
	id = GSL_focus
	country = {
		factor = 0
		modifier = {
			OR = {
				has_country_flag = is_slaver_nation_flag
				original_tag = VRX
				original_tag = JAD
				original_tag = TOL
				original_tag = EXC
				original_tag = BFS
				original_tag = WHJ
				original_tag = MAW
				original_tag = NII
			}
			add = 20
		}
	}
	default = no
	reset_on_civilwar = no
	continuous_focus_position = {
		x = 3300
		y = 900
	}
	shared_focus = GSL_where_the_chains_came_from
	shared_focus = GSL_who_holds_the_whip
	shared_focus = GSL_guns_for_the_overseers
}
