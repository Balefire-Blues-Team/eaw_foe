# Missile destroyer hull: reachable by nobody before this project
sp_naval_missile_destroyer = {
	specialization = specialization_naval
	special_project_parent = {
		sp_naval_bombardment_ship
	}
	project_tags = sp_tag_surface_ship
	prototype_time = sp_time.prototype.medium
	complexity = sp_complexity.medium

	breakthrough_cost = {
		specialization_naval = 1
	}

	visible = {
		FROM = {
			OR = {
				NOT = { has_tech = foe_missile_destroyer_hull_tech }
				is_special_project_completed = sp:sp_naval_missile_destroyer
			}
		}
	}
	available = {
		FROM = {
			has_tech = foe_light_navy_hull_tech_5
		}
	}

	ai_will_do = {
		base = 5
	}

	project_output = {
		country_effects = {
			set_technology = {
				foe_missile_destroyer_hull_tech = 1
				popup = no
			}
		}
	}
}
