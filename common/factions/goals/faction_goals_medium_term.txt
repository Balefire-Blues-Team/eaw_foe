faction_goal_market_domination = {
	name = market_domination_goal_name
	description = market_domination_goal_desc
	group = FOCUS_FILTER_INDUSTRY
	category = medium_term
	visible = {
		always = yes
	}
	select_effect = {
		set_temp_variable = { total_nodes = 0 }
		every_collection_element = {
			input = {
				input = game:scope
				operators =  {
					faction_members
				}
			}
			add_to_temp_variable = { total_nodes = state_controlled_nodes^num }
		}

		multiply_temp_variable = {
			var = total_nodes
			value = 1.50
		}
		clamp_temp_variable = {
			var = total_nodes
			min = 1
		}
		scope:faction = {
			set_variable = { unlock_trade_nodes_target = total_nodes }
		}
	}
	completed = {
		scope:faction = {
			hidden_trigger = {
				tooltip_evaluation = eval
				set_temp_variable = { unlock_trade_nodes_compare = 0 }
				all_collection_elements = {
					collection = {
						input = game:scope
						operators =  { faction_members }
					}
					add_to_temp_variable = { unlock_trade_nodes_compare = state_controlled_nodes^num }
				}
				set_temp_variable = { unlock_trade_nodes_compare_temp_target = unlock_trade_nodes_compare }
				multiply_temp_variable = { unlock_trade_nodes_compare_temp_target = 1.50 }
			}
			if = {
				limit = {
					has_variable = unlock_trade_nodes_target
				}
				check_variable = {
					var = unlock_trade_nodes_compare
					value = unlock_trade_nodes_target
					compare = greater_than
					tooltip = "GOAL_NODES_TRIGGER_A"
				}
			}
			else = {
				check_variable = {
					var = unlock_trade_nodes_compare
					value = unlock_trade_nodes_compare_temp_target
					compare = greater_than
					tooltip = "GOAL_NODES_TRIGGER_A"
				}
			}
		}
	}
	complete_effect = {
		add_faction_power_projection = 100
		custom_effect_tooltip = generic_skip_one_line_tt
		custom_effect_tooltip = ALL_FACTION_MEMBERS_TRADE_NODES
		every_faction_member = {
			limit = {
				check_variable = {
					state_controlled_nodes^num > 0
				}
			}
			add_faction_initiative = 1
		}
		custom_effect_tooltip = generic_skip_one_line_tt
		custom_effect_tooltip = ALL_FACTION_MEMBERS
		every_faction_member = {
			set_temp_variable = { caps_to_add = 25 }
			add_caps = yes
			set_country_flag = faction_trade_completed
		}
	}
	ai_will_do = {
		base = 1
	}
}
