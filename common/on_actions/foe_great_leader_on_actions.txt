on_actions = {
	# ROOT = new controller, FROM = previous controller, FROM.FROM = state.
	# Remove immediately on court loss rather than waiting for idea cancellation.
	
	#Nomadic Khan actions
	on_state_control_changed = {
		effect = {
			FROM = {
				if = {
					limit = {
						has_idea = FOE_the_court_is_the_state
						NOT = { FOE_nomadic_court_controlled = yes }
					}
					remove_ideas = FOE_the_court_is_the_state
				}
			}
		}
	}

	#Slaving Despot actions
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					FOE_is_slaving_despot = yes
					has_war_with = FROM
					FROM.FROM = {
						NOT = { is_core_of = ROOT }
						NOT = { has_state_flag = FOE_despot_conquest_slaves_taken }
					}
				}
				FOE_take_conquest_slaves = yes
			}
		}
	}

	on_monthly = {
		effect = {
			if = {
				limit = {
					OR = {
						FOE_is_mythic_slaving_despot = yes
						has_dynamic_modifier = { modifier = FOE_empire_of_chains_modifier }
					}
				}
				FOE_update_empire_of_chains = yes
			}
		}
	}

	#Carrion Marshal actions
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					FOE_is_mythic_carrion_marshal = yes
					has_war_with = FROM
					FROM.FROM = {
						NOT = { is_owned_by = ROOT }
						FOE_is_harvestable_state = yes
					}
				}
				FROM.FROM = { FOE_carrion_harvest_state = yes }
			}
		}
	}

	on_monthly = {
		effect = {
			FOE_update_living_off_the_land = yes
		}
	}

	#Unifier actions
	on_monthly = {
		effect = {
			FOE_update_one_nation = yes
		}
	}

	#Puppetmaster actions
	on_monthly = {
		effect = {
			if = {
				limit = {
					OR = {
						FOE_is_legendary_or_mythic_puppetmaster = yes
						has_dynamic_modifier = { modifier = FOE_web_of_strings_modifier }
					}
				}
				FOE_update_puppet_web = yes
			}
		}
	}

	#Architect of Order actions
	on_state_control_changed = {
		effect = {
			FROM = {
				if = {
					limit = {
						FOE_is_mythic_architect_of_order = yes
						NOT = { has_country_flag = FOE_centre_fallen }
						has_variable = FOE_architect_heart_state
						var:FOE_architect_heart_state = { NOT = { is_controlled_by = PREV } }
					}
					set_country_flag = FOE_centre_fallen
					add_ideas = FOE_the_centre_has_fallen
					FOE_update_planned_capital = yes
				}
			}
			if = {
				limit = {
					has_country_flag = FOE_centre_fallen
					has_variable = FOE_architect_heart_state
					var:FOE_architect_heart_state = { is_controlled_by = PREV }
				}
				clr_country_flag = FOE_centre_fallen
				remove_ideas = FOE_the_centre_has_fallen
				FOE_update_planned_capital = yes
			}
		}
	}

	on_monthly = {
		effect = {
			if = {
				limit = {
					OR = {
						FOE_is_architect_of_order = yes
						has_variable = FOE_planned_capital_state
					}
				}
				FOE_update_planned_capital = yes
			}
		}
	}

	#Reactor Prophet actions
	on_monthly = {
		effect = {
			if = {
				limit = {
					OR = {
						FOE_is_reactor_prophet = yes
						has_dynamic_modifier = { modifier = FOE_reactor_cult_modifier }
					}
				}
				FOE_update_reactor_prophet = yes
			}
		}
	}

	#Autark actions
	on_monthly = {
		effect = {
			if = {
				limit = {
					OR = {
						FOE_is_autark = yes
						has_dynamic_modifier = { modifier = FOE_autarky_modifier }
					}
				}
				FOE_update_autark = yes
			}
		}
	}
}
