on_actions = {
	# Skip subclass work when Great Leaders were disabled in campaign setup.
	# Keep the gates outside scope changes and preserve cleanup checks within enabled campaigns.
	# Each new incumbent gets one chance, even after a same-party scripted succession.
	# The character marker also remembers failed rolls and returning former leaders.
	on_weekly = {
		effect = {
			# Do not attempt succession checks unless the campaign enabled Great Leaders.
			if = {
				limit = {
					has_global_flag = FOE_campaign_great_leaders_enabled
					has_global_flag = FOE_campaign_setup_confirmed
					NOT = { has_global_flag = FOE_campaign_setup_multiplayer }
				}
				FOE_check_great_leader_succession = yes
			}
		}
	}

	# ROOT = new controller, FROM = previous controller, FROM.FROM = state.
	# Remove immediately on court loss rather than waiting for idea cancellation.
	
	#Nomadic Khan actions
	on_state_control_changed = {
		effect = {
			if = {
				limit = { has_global_flag = FOE_campaign_great_leaders_enabled }
				FROM = {
					if = {
						limit = {
							has_idea = FOE_the_court_is_the_state
							NOT = { FOE_nomadic_court_controlled = yes }
						}
						remove_ideas = FOE_the_court_is_the_state
					}
				}
			}
		}
	}

	# ROOT is the country that capitulated. Only a capitulation during the Reckoning, while the two are at war, counts.
	on_capitulation = {
		effect = {
			if = {
				limit = {
					has_global_flag = FOE_campaign_great_leaders_enabled
					FOE_has_active_reckoning = yes
					has_country_flag = FOE_nemesis_war_started
				}
				var:FOE_nemesis_rival = {
					if = {
						limit = { has_war_with = PREV }
						set_country_flag = FOE_nemesis_rival_fell
						PREV = { set_country_flag = FOE_nemesis_we_fell }
					}
				}
			}
		}
	}

	# THIS is the operation, ROOT the initiating country, FROM the target.
	# An ordinary operation against the Black File adds field experience; the Broker's own two do not refund themselves.
	on_operation_completed = {
		effect = {
			if = {
				limit = {
					has_global_flag = FOE_campaign_great_leaders_enabled
					NOT = { is_operation_type = FOE_operation_open_ledger }
					NOT = { is_operation_type = FOE_operation_crossed_wires }
				}
				ROOT = {
					if = {
						limit = {
							FOE_is_whisper_broker = yes
							has_variable = FOE_black_file
							var:FOE_black_file = { tag = FROM }
							NOT = { has_country_flag = FOE_broker_alerted }
						}
						add_to_variable = { FOE_dossier = 1 }
						clamp_variable = { var = FOE_dossier min = 0 max = 6 }
						FOE_refresh_insider_knowledge = yes
					}
				}
			}
		}
	}
