on_actions = {
	# Antagonist rubber band, expansion, Contact and first strike
	on_monthly = {
		effect = {
			if = {
				limit = {
					has_global_flag = FOE_campaign_antagonists_enabled
					has_country_flag = FOE_antagonist
				}
				FOE_update_antagonist_rubber_band = yes
				FOE_update_antagonist_expansion = yes
				FOE_update_antagonist_confrontation = yes
			}
		}
	}

	on_peaceconference_ended = {
		effect = {
			if = {
				limit = {
					has_global_flag = FOE_campaign_antagonists_enabled
					has_country_flag = FOE_antagonist
				}
				set_variable = { FOE_antagonist_stall_months = 0 }
				set_variable = { FOE_antagonist_nudge_months = 0 }
			}
		}
	}
}
