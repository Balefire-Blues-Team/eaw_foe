	on_monthly = {
		effect = {
			if = {
				limit = {
					has_country_flag = slavery_allowed
				}
				GSL_market_monthly = yes

	# Generic slaver tree, Spoils of War: ROOT is the new controller, FROM.FROM the state
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					has_completed_focus = GSL_spoils_of_war
					FROM.FROM = {
						NOT = { is_core_of = ROOT }
						NOT = { has_state_flag = GSL_spoils_taken_flag }
						state_population > 2999
					}
				}
				FROM.FROM = {
					set_state_flag = GSL_spoils_taken_flag
				}

			}
