	#Builds a random resource generator in a core state
	COM_autarky_secure_supply = {
		icon = GFX_decision_stable_door
		state_target = yes
		target_array = owned_states
		on_map_mode = map_and_decisions_view
		cost = 25
		fire_only_once = no
		days_remove = 90
		visible = { original_tag = COM }
		target_root_trigger = { original_tag = COM }
		target_trigger = {
			FROM = { COM_is_autarky_supply_site = yes }
		}
		available = {
			custom_trigger_tooltip = {
				tooltip = COM_autarky_supply_cooldown_tt
				NOT = { has_country_flag = COM_autarky_supply_cooldown_flag }
			}
			FROM = { COM_is_autarky_supply_site = yes }
		}
		modifier = {
			civilian_factory_use = 5
		}
		complete_effect = {
			hidden_effect = {
				if = { #Substitutes for Everything halves the wait
					limit = { has_country_flag = COM_import_substitution_flag }
					set_country_flag = { flag = COM_autarky_supply_cooldown_flag value = 1 days = 90 }
				}
				else = {
					set_country_flag = { flag = COM_autarky_supply_cooldown_flag value = 1 days = 180 }
				}
			}
		}
		remove_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision COM_autarky_secure_supply in [From.GetName]"
			custom_effect_tooltip = COM_autarky_secure_supply_tt
			hidden_effect = {
				COM_autarky_build_random_generator = yes
				COM_update_autarky = yes
			}
		}
		ai_will_do = {
			base = 20
			modifier = {
				factor = 0
				num_of_available_civilian_factories < 6
			}
		}
	}
