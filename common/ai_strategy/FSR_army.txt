### Army size

# Floor: eight divisions are the minimum to cover FSR's frontline.
FSR_army_floor = {
	enable = {
		original_tag = FSR
		num_divisions < 8
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = force_build_armies
		value = 300
	}
}

# Cap: 8 + military factories (x1.5 at war, x1.2 at each of 20/40/60% surrender progress, x0.7 on easy against a
# human). Above it, factories go to equipment instead of new divisions.
FSR_army_cap = {
	enable = {
		original_tag = FSR
		set_temp_variable = { division_count = num_of_military_factories }
		add_to_temp_variable = { division_count = 8 }
		if = {
			limit = { has_war = yes }
			multiply_temp_variable = { division_count = 1.5 }
		}
		if = {
			limit = { surrender_progress > 0.2 }
			multiply_temp_variable = { division_count = 1.2 }
		}
		if = {
			limit = { surrender_progress > 0.4 }
			multiply_temp_variable = { division_count = 1.2 }
		}
		if = {
			limit = { surrender_progress > 0.6 }
			multiply_temp_variable = { division_count = 1.2 }
		}
		if = {
			limit = {
				difficulty < 2
				any_country = {
					has_war_with = ROOT
					is_ai = no
				}
			}
			multiply_temp_variable = { division_count = 0.7 }
		}
		check_variable = { num_divisions > division_count }
	}
	abort_when_not_enabled = yes

	# every role FSR_unit_production (FSR.txt) asks for
	ai_strategy = {
		type = build_army
		id = power_armour
		value = -500
	}
	ai_strategy = {
		type = build_army
		id = infantry
		value = -500
	}
	ai_strategy = {
		type = build_army
		id = garrison
		value = -500
	}
	ai_strategy = {
		type = build_army
		id = armor
		value = -500
	}
	ai_strategy = {
		type = build_army
		id = support_robot
		value = -500
	}
	ai_strategy = {
		type = build_army
		id = combat_robot
		value = -500
	}
}

# Special forces room: once fewer than 10 special-forces battalions are free, stop building power armour and let
# regular units raise the cap, instead of squeezing in 2-3 battalion PA divisions (user test, 2026-09-28).
# Cap as the engine describes it (defines SPECIAL_FORCES_CAP_BASE 0.05, SPECIAL_FORCES_CAP_MIN 24):
# max(24 + special_forces_min, regular battalions x 0.05 x (1 + special_forces_cap)) + special_forces_cap_flat.
FSR_special_forces_full = {
	enable = {
		original_tag = FSR
		set_temp_variable = { sf_used = num_battalions_with_type@power_armour }
		add_to_temp_variable = { sf_used = num_battalions_with_type@power_armour_fireteam_infantry }
		add_to_temp_variable = { sf_used = num_battalions_with_type@power_armour_demolitions_infantry }
		add_to_temp_variable = { sf_used = num_battalions_with_type@hq_power_armour }
		set_temp_variable = { sf_cap_share = modifier@special_forces_cap }
		add_to_temp_variable = { sf_cap_share = 1 }
		multiply_temp_variable = { sf_cap_share = 0.05 }
		set_temp_variable = { sf_cap = num_battalions }
		subtract_from_temp_variable = { sf_cap = sf_used }
		multiply_temp_variable = { sf_cap = sf_cap_share }
		set_temp_variable = { sf_cap_min = modifier@special_forces_min }
		add_to_temp_variable = { sf_cap_min = 24 }
		if = {
			limit = { check_variable = { sf_cap < sf_cap_min } }
			set_temp_variable = { sf_cap = sf_cap_min }
		}
		add_to_temp_variable = { sf_cap = modifier@special_forces_cap_flat }
		subtract_from_temp_variable = { sf_cap = sf_used }
		check_variable = { sf_cap < 10 }
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = build_army
		id = power_armour
		value = -500
	}
}

### Reserve, garrison, supply

# Keep a fifth of the army in the capital (FSR's only port) against invasions, while the fronts do not already
# cover it. Only against naval powers: in a land war the reserve just stood idle (user test, 2026-09-28).
FSR_reserve_capital = {
	enable = {
		original_tag = FSR
		has_war = yes
		any_enemy_country = {
			has_navy_size = { size > 10 }
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = put_unit_buffers
		ratio = 0.2
		states = {
			538
		}
		subtract_fronts_from_need = yes
	}
}
