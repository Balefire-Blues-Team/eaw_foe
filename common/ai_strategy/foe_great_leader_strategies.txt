	allowed = { always = yes }
	enable = {
		is_ai = yes

			FOE_is_architect_of_order = yes
			AND = {
				FOE_is_mythic_nomadic_khan = yes
				has_variable = FOE_nomadic_court_state
			}
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = front_unit_request
		state_trigger = {
			OR = {
				FOE_ai_is_critical_seat = yes
				any_neighbor_state = { FOE_ai_is_critical_seat = yes }
			}
		}
		value = 50
	}
}
