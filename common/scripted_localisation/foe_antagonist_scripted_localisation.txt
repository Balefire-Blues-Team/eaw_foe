# Antagonist System tracker. Called as [Root.GetFOE<Ring><Part>] from the player's decisions; each reads its
# Antagonist through var:global, so a fallen (no longer existing) Antagonist is caught first.

defined_text = {
	name = GetFOERivalStatus
	text = {
		trigger = { has_global_flag = FOE_antagonist_defeated_1 }
		localization_key = FOE_tracker_status_defeated
	}
	text = {
		trigger = { has_global_flag = FOE_antagonist_fallen_slot_1 }
		localization_key = FOE_tracker_status_fallen
	}
	text = {
		trigger = { var:global.FOE_antagonist_rival = { exists = no } }
		localization_key = FOE_tracker_status_fallen
	}
	text = {
		trigger = { var:global.FOE_antagonist_rival = { check_variable = { FOE_antagonist_status = 4 } } }
		localization_key = FOE_tracker_status_war
	}
	text = {
		trigger = { var:global.FOE_antagonist_rival = { check_variable = { FOE_antagonist_status = 5 } } }
		localization_key = FOE_tracker_status_preparing
	}
	text = {
		trigger = { var:global.FOE_antagonist_rival = { check_variable = { FOE_antagonist_status = 3 } } }
		localization_key = FOE_tracker_status_ready
	}
	text = {
		trigger = { var:global.FOE_antagonist_rival = { check_variable = { FOE_antagonist_status = 2 } } }
		localization_key = FOE_tracker_status_border
	}
	text = {
		localization_key = FOE_tracker_status_consolidating
	}
}
