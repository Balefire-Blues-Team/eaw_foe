### Severyanan Reconstruction Authority ###

# State scope, ROOT is the Authority: the state becomes a core, its new citizens wait in the census queue
SEZ_integrate_state = {
	add_core_of = ROOT
	if = {
		limit = { ROOT = { has_country_flag = SEZ_reconciliation_flag } }
		add_dynamic_modifier = {
			modifier = SEZ_census_backlog_modifier
			days = 180
		}
	}
	else = {
		add_dynamic_modifier = {
			modifier = SEZ_census_backlog_modifier
			days = 365
		}
	}
}

# State scope, ROOT is the Authority
SEZ_spawn_garrison_robots = {
	create_unit = {
		division = "name = \"Garrison Robots\" division_template = \"Garrison Robots\" start_experience_factor = 0.2"
		owner = ROOT
	}
}

# The AI cannot use the Princessyn screen, so its Upgrade Points are spent for it
SEZ_ai_place_upgrades = {
	if = {
		limit = { is_ai = yes }
		every_owned_state = {
			limit = { check_variable = { SEZ_build_points > 0 } }
			add_to_variable = { SEZ_industry_dm = 0.10 }
			add_to_variable = { SEZ_militarization_dm = 0.10 }
			add_building_construction = {
				type = industrial_complex
				level = 1
			}
			set_variable = { SEZ_build_points = 0 }
		}
	}
}
