# Country scope; FROM is the target state. Builds one level of a random generator that still fits.
COM_autarky_build_random_generator = {
	random_list = {
		log = yes
		20 = {
			modifier = {
				factor = 0
				FROM = { NOT = { free_building_slots = { building = water_generator size > 0 } } }
			}
			FROM = { add_building_construction = { type = water_generator level = 1 instant_build = yes } }
		}
		20 = {
			modifier = {
				factor = 0
				FROM = { NOT = { free_building_slots = { building = electricity_generator size > 0 } } }
			}
			FROM = { add_building_construction = { type = electricity_generator level = 1 instant_build = yes } }
		}
		20 = {

			has_country_flag = COM_power_struggle_flag

# Country scope (the Hegemony). Turns qualifying subjects into zones, and releases qualifying countries
# that no longer exist as zones on whatever of their core land we own.
COM_establish_purification_zones = {
	every_country = {
		limit = {
			is_subject_of = ROOT
			COM_become_purification_zone = yes
			NOT = { has_autonomy_state = autonomy_purification_zone }
		}
		ROOT = { set_autonomy = { target = PREV autonomy_state = autonomy_purification_zone } }
		COM_set_purification_zone_cosmetic = yes
	}
	every_possible_country = {
		limit = {
			exists = no
			COM_become_purification_zone = yes
			any_core_state = { is_owned_by = ROOT }
		}
		ROOT = { release_autonomy = { target = PREV autonomy_state = autonomy_purification_zone } }
		COM_set_purification_zone_cosmetic = yes
	}
}
