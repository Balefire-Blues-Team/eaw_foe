foe_unlock_CAL_char_constituto = {
	if = {
		limit = { NOT = { CAL_char_constituto = { has_character_flag = foe_roles_unlocked } } }
		add_advisor_role = {
			character = CAL_char_constituto
			advisor = {
				slot = army_chief
				idea_token = CAL_constituto
				allowed = {
					original_tag = CAL
				}
				traits = {
					drill_seargent_trait
				}
				cost = 150
				ai_will_do = {
					factor = 90
				}
			}
		}
		CAL_char_constituto = { set_character_flag = foe_roles_unlocked }
	}
}
