### Generic slaver tree (2026-10-14): the house, the gangs and The Pens, the slave market, expeditions ###
### Slaves are counted in thousands: foe_slaves_var holds the free ones, GSL_gang_<type> the ones held in gangs.

# Country scope: defaults of the slave system, from the first history focus
GSL_init_house = {
	add_dynamic_modifier = {
		modifier = GSL_the_house_modifier
	}
	set_variable = { GSL_pens_cost_mult = 1 }
	set_variable = { GSL_gang_bonus_mult = 1 }
	set_variable = { GSL_levy_mult = 1 }
	set_variable = { GSL_hold_cap = 4 }
	set_variable = { GSL_event_mult = 1 }
	GSL_init_market = yes
}

# Country scope: recomputes what the slaves in the gangs give and cost
GSL_update_pens = {
	set_variable = { GSL_held = GSL_gang_field }
	add_to_variable = { GSL_held = GSL_gang_factory }
	add_to_variable = { GSL_held = GSL_gang_road }
	add_to_variable = { GSL_held = GSL_gang_levy }
	set_temp_variable = { cost_mult = GSL_pens_cost_mult }
	clamp_temp_variable = { var = cost_mult min = 0.1 }
	set_variable = { GSL_pens_stability = GSL_held }
	multiply_variable = { GSL_pens_stability = cost_mult }
	multiply_variable = { GSL_pens_stability = -0.02 }
	set_variable = { GSL_pens_research = GSL_held }
	multiply_variable = { GSL_pens_research = cost_mult }
	multiply_variable = { GSL_pens_research = -0.01 }
	set_variable = { GSL_pens_metal = GSL_gang_field }
	multiply_variable = { GSL_pens_metal = GSL_gang_bonus_mult }
	set_variable = { GSL_pens_energy = GSL_pens_metal }
	set_variable = { GSL_pens_output = GSL_gang_factory }
	multiply_variable = { GSL_pens_output = GSL_gang_bonus_mult }
	multiply_variable = { GSL_pens_output = 0.03 }
	set_variable = { GSL_pens_construction = GSL_gang_road }
	multiply_variable = { GSL_pens_construction = GSL_gang_bonus_mult }
	multiply_variable = { GSL_pens_construction = 0.04 }
	set_variable = { GSL_pens_recruit = GSL_gang_levy }
	multiply_variable = { GSL_pens_recruit = GSL_gang_bonus_mult }
	multiply_variable = { GSL_pens_recruit = GSL_levy_mult }
	multiply_variable = { GSL_pens_recruit = 0.03 }
	force_update_dynamic_modifier = yes
}

# Country scope: one thousand held slaves are lost (runaways, ringleaders), from the largest gang
GSL_lose_held_slave = {
	if = {
		limit = {
			check_variable = { GSL_held > 0 }
		}
		if = {
			limit = {
				check_variable = { GSL_gang_field > 0 }
				check_variable = { GSL_gang_field > GSL_gang_factory }
				check_variable = { GSL_gang_field > GSL_gang_road }
				check_variable = { GSL_gang_field > GSL_gang_levy }
			}
			subtract_from_variable = { GSL_gang_field = 1 }
		}
		else_if = {
			limit = {
				check_variable = { GSL_gang_factory > 0 }
				check_variable = { GSL_gang_factory > GSL_gang_road }
				check_variable = { GSL_gang_factory > GSL_gang_levy }
			}
			subtract_from_variable = { GSL_gang_factory = 1 }
		}
		else_if = {
			limit = {
				check_variable = { GSL_gang_road > 0 }
				check_variable = { GSL_gang_road > GSL_gang_levy }
			}
			subtract_from_variable = { GSL_gang_road = 1 }
		}
		else_if = {
			limit = {
				check_variable = { GSL_gang_levy > 0 }
			}
			subtract_from_variable = { GSL_gang_levy = 1 }
		}
		else_if = {
			limit = {
				check_variable = { GSL_gang_field > 0 }
			}
			subtract_from_variable = { GSL_gang_field = 1 }
		}
		else_if = {
			limit = {
				check_variable = { GSL_gang_factory > 0 }
			}
			subtract_from_variable = { GSL_gang_factory = 1 }
		}
		else = {
			subtract_from_variable = { GSL_gang_road = 1 }
		}
		GSL_update_pens = yes
	}
}

# Country scope: monthly upkeep of The Pens: one flavour event at most every 90 days while 3 thousand or more are held.
# Chance per month: 3% per thousand held, times GSL_event_mult (Works x1.5, Market x0.5, Overseer Corps x0.5)
GSL_pens_monthly = {
	if = {
		limit = {
			has_dynamic_modifier = { modifier = GSL_the_pens_modifier }
			check_variable = { GSL_held > 2.9 }
			NOT = { has_country_flag = GSL_pens_event_cooldown_flag }
		}
		set_temp_variable = { chance = GSL_held }
		multiply_temp_variable = { chance = 3 }
		multiply_temp_variable = { chance = GSL_event_mult }
		set_temp_variable_to_random = { var = roll min = 0 max = 100 }
		if = {
			limit = {
				check_variable = { roll < chance }
			}
			set_country_flag = { flag = GSL_pens_event_cooldown_flag days = 90 value = 1 }
			random_list = {
				40 = { country_event = { id = GSL.1 } }
				30 = { country_event = { id = GSL.2 } }
				30 = { country_event = { id = GSL.3 } }
			}
		}
	}
}

# State scope (a state of PREV): the state loses 1000 inhabitants without touching its owner's manpower pool
GSL_take_population = {
	OWNER = {
		save_current_manpower = yes
	}
	add_manpower = -1000
	OWNER = {
		restore_previous_manpower = yes
	}
}

### Slave market. global.GSL_price is the going price of one thousand slaves in Economic Capacity Surplus,
### global.GSL_pool the slaves on offer. Sellers get the price times GSL_sell_mult, buyers pay 20% more times GSL_buy_mult.

# State scope (a state of PREV): 1000 freed slaves settle in the state, without touching its owner's manpower pool
GSL_settle_population = {
	OWNER = {
		save_current_manpower = yes
	}
	add_manpower = 1000
	OWNER = {
		restore_previous_manpower = yes
	}
}

# Country scope: market defaults (any slaver nation, not only the tree's)
GSL_init_market = {
	if = {
		limit = {
			NOT = { has_variable = GSL_sell_mult }
		}
		set_variable = { GSL_sell_mult = 1 }
		set_variable = { GSL_buy_mult = 1 }
		# Old Appleloosa, the centre of the slave trade
		if = {
			limit = { has_idea = OAP_centre_of_slave_trade }
			add_to_variable = { GSL_sell_mult = 0.1 }
			add_to_variable = { GSL_buy_mult = -0.1 }
		}
		else_if = {
			limit = { has_idea = OAP_centre_of_slave_trade2 }
			add_to_variable = { GSL_sell_mult = 0.2 }
			add_to_variable = { GSL_buy_mult = -0.2 }
		}
		# Red Eye: Fillydelphia pays any price
		if = {
			limit = { original_tag = EYE }
			add_to_variable = { GSL_buy_mult = 0.5 }
		}
	}
}

# Any country scope: starting market, from on_startup (and the first monthly tick of an older save)
GSL_init_global_market = {
	if = {
		limit = {
			NOT = { has_global_flag = GSL_market_started_flag }
		}
		set_global_flag = GSL_market_started_flag
		set_variable = { global.GSL_price = 600 }
		set_variable = { global.GSL_pool = 10 }
		set_variable = { global.GSL_base_price = 600 }
		GSL_reprice = yes
	}
}

# Global: the prices in political power, guns and factory days that follow the going price
GSL_reprice = {
	# saves from before the normal price existed
	if = {
		limit = {
			check_variable = { global.GSL_base_price < 1 }
		}
		set_variable = { global.GSL_base_price = 600 }
	}
	# the price stays between half and double the normal price
	set_temp_variable = { low = global.GSL_base_price }
	multiply_temp_variable = { low = 0.5 }
	set_temp_variable = { high = global.GSL_base_price }
	multiply_temp_variable = { high = 2 }
	clamp_variable = { var = global.GSL_price min = low max = high }
	set_variable = { global.GSL_autorenew_limit = global.GSL_base_price }
	multiply_variable = { global.GSL_autorenew_limit = 1.25 }
	round_variable = global.GSL_autorenew_limit
	set_variable = { global.GSL_sell_pp = global.GSL_price }
	multiply_variable = { global.GSL_sell_pp = 0.0417 }
	set_variable = { global.GSL_sell_guns = global.GSL_price }
	multiply_variable = { global.GSL_sell_guns = 0.1667 }
	set_variable = { global.GSL_buy_pp = global.GSL_price }
	multiply_variable = { global.GSL_buy_pp = 0.0667 }
	set_variable = { global.GSL_buy_guns = global.GSL_price }
	multiply_variable = { global.GSL_buy_guns = 0.2083 }
	set_variable = { global.GSL_buy_days = global.GSL_price }
	multiply_variable = { global.GSL_buy_days = 0.24 }
	round_variable = global.GSL_sell_pp
	round_variable = global.GSL_sell_guns
	round_variable = global.GSL_buy_pp
	round_variable = global.GSL_buy_guns
	round_variable = global.GSL_buy_days
}

# Country scope: a batch was sold to the market (the slaves are already taken off the seller)
GSL_market_sold = {
	add_to_variable = { global.GSL_pool = 1 }
	multiply_variable = { global.GSL_price = 0.95 }
	GSL_reprice = yes
	if = {
		limit = {
			has_completed_focus = GSL_payment_on_delivery
		}
		add_political_power = 10
	}
}

# Country scope: a batch was bought from the market
GSL_market_bought = {
	subtract_from_variable = { global.GSL_pool = 1 }
	multiply_variable = { global.GSL_price = 1.05 }
	GSL_reprice = yes
}

# Country scope: pays temp variable surplus into the bank (add_cic in steps of 10)
GSL_pay_surplus = {
	set_temp_variable = { steps = surplus }
	divide_temp_variable = { steps = 10 }
	round_temp_variable = steps
	while_loop_effect = {
		limit = {
			check_variable = { steps > 0 }
		}
		add_cic = 10
		subtract_from_temp_variable = { steps = 1 }
	}
}

# Country scope: adds temp variable days to our market contract (one civilian factory tied up until paid off)
GSL_add_debt = {
	add_to_variable = { GSL_debt_days = days }
	if = {
		limit = {
			NOT = { has_dynamic_modifier = { modifier = GSL_market_debt_modifier } }
		}
		add_dynamic_modifier = {
			modifier = GSL_market_debt_modifier
		}
	}
}

# Global, once a month: the price drifts 10% of the way back to the normal price and a little stock arrives from far away
GSL_market_monthly = {
	# saves started before the market existed set it up here
	GSL_init_global_market = yes
	if = {
		limit = {
			check_variable = { global.GSL_base_price < 1 }
		}
		set_variable = { global.GSL_base_price = 600 }
	}
	if = {
		limit = {
			NOT = { has_global_flag = GSL_market_drift_flag }
		}
		set_global_flag = { flag = GSL_market_drift_flag days = 25 value = 1 }
		set_temp_variable = { gap = global.GSL_base_price }
		subtract_from_temp_variable = { gap = global.GSL_price }
		multiply_temp_variable = { gap = 0.1 }
		add_to_variable = { global.GSL_price = gap }
		if = {
			limit = {
				check_variable = { global.GSL_pool < 10 }
			}
			add_to_variable = { global.GSL_pool = 0.5 }
		}
		GSL_reprice = yes
		GSL_market_watch = yes
	}
}

### Market hooks: call these from any event, focus, decision or on_action (any country scope) ###

# A shock moves the price and the pool now; the monthly drift erodes it again (about half is gone after 7 months).
# Set temp variables before the call: shock_price (factor, e.g. 1.25 or 0.85), shock_pool (thousands, may be negative)
GSL_market_shock = {
	multiply_variable = { global.GSL_price = shock_price }
	add_to_variable = { global.GSL_pool = shock_pool }
	clamp_variable = { var = global.GSL_pool min = 0 }
	GSL_reprice = yes
}

# A shift moves the normal price the drift returns to, for good (until another shift).
# Set temp variable shift_price (added, e.g. 150 or -100) before the call. The normal price stays between 300 and 1200.
GSL_market_shift = {
	add_to_variable = { global.GSL_base_price = shift_price }
	clamp_variable = { var = global.GSL_base_price min = 300 max = 1200 }
	GSL_reprice = yes
}

# Every slaver nation hears about a market event (temp variable market_event = number of the GSL.2xx event)
GSL_market_notice = {
	every_country = {

			has_country_flag = slavery_allowed
			NOT = { original_tag = RWK }
		}
		if = {
			limit = { check_variable = { market_event = 201 } }
			country_event = { id = GSL.201 }
		}
		else_if = {
			limit = { check_variable = { market_event = 202 } }
			country_event = { id = GSL.202 }
		}
		else_if = {
			limit = { check_variable = { market_event = 203 } }
			country_event = { id = GSL.203 }
		}
		else_if = {
			limit = { check_variable = { market_event = 204 } }
			country_event = { id = GSL.204 }
		}
		else_if = {
			limit = { check_variable = { market_event = 205 } }
			country_event = { id = GSL.205 }
		}
	}
}

# Global, monthly: market events that are watched for rather than fired by their cause
GSL_market_watch = {
	# Fillydelphia's industry passes 100 factories: lasting demand for hands
	if = {
		limit = {
			NOT = { has_global_flag = GSL_eye_industry_boom_flag }
			country_exists = EYE
			EYE = {
				num_of_factories > 99
			}
		}
		set_global_flag = GSL_eye_industry_boom_flag
		set_temp_variable = { shift_price = 150 }
		GSL_market_shift = yes
		set_temp_variable = { shock_price = 1.1 }
		set_temp_variable = { shock_pool = 0 }
		GSL_market_shock = yes
		set_temp_variable = { market_event = 201 }
		GSL_market_notice = yes
	}
}

# Country scope (the buyer), monthly: renews a paid-off contract when auto-renew is on and the price is acceptable
GSL_autorenew_monthly = {
	if = {
		limit = {
			has_country_flag = GSL_autorenew_flag
			NOT = { has_dynamic_modifier = { modifier = GSL_market_debt_modifier } }
		}
		if = {
			limit = {
				check_variable = { global.GSL_price > global.GSL_autorenew_limit }
			}
			set_country_flag = GSL_autorenew_paused_flag
		}
		else = {
			clr_country_flag = GSL_autorenew_paused_flag
			if = {
				limit = {
					GSL_can_buy_factories = yes
				}
				GSL_trade_buy_factories = yes
			}
		}
	}
	else = {
		clr_country_flag = GSL_autorenew_paused_flag
	}
}

# Country scope, monthly: contracts, standing orders, debts, tribute
GSL_country_monthly = {
	if = {
		limit = {
			has_dynamic_modifier = { modifier = GSL_market_debt_modifier }
		}
		subtract_from_variable = { GSL_debt_days = 30 }
		if = {
			limit = {
				check_variable = { GSL_debt_days < 1 }
			}
			set_variable = { GSL_debt_days = 0 }
			remove_dynamic_modifier = { modifier = GSL_market_debt_modifier }
		}
	}
	GSL_autorenew_monthly = yes
	if = {
		limit = {
			has_dynamic_modifier = { modifier = GSL_standing_order_modifier }
		}
		if = {
			limit = {
				check_variable = { GSL_order_left > 0 }
			}
			add_to_variable = { foe_slaves_var = 1 }
			subtract_from_variable = { GSL_order_left = 1 }
		}
		subtract_from_variable = { GSL_order_days = 30 }
		if = {
			limit = {
				check_variable = { GSL_order_days < 1 }
				check_variable = { GSL_order_left < 1 }
			}
			remove_dynamic_modifier = { modifier = GSL_standing_order_modifier }
		}
	}
	if = {
		limit = {
			OR = {
				has_completed_focus = GSL_debts_paid_in_years
				has_completed_focus = GSL_tribute_in_flesh
			}
		}
		add_to_variable = { GSL_quarter = 1 }
		if = {
			limit = {
				check_variable = { GSL_quarter > 2 }
			}
			set_variable = { GSL_quarter = 0 }
			if = {
				limit = {
					has_completed_focus = GSL_debts_paid_in_years
					capital_scope = { state_population > 2000 }
				}
				add_to_variable = { foe_slaves_var = 1 }
				capital_scope = { GSL_take_population = yes }
			}
			if = {
				limit = {
					has_completed_focus = GSL_tribute_in_flesh
				}
				every_subject_country = {
					limit = {
						capital_scope = { state_population > 2000 }
					}
					ROOT = { add_to_variable = { foe_slaves_var = 1 } }
					capital_scope = { GSL_take_population = yes }
				}
			}
		}
	}
	GSL_pens_monthly = yes
}

### Abolition: shared by every nation that ends slavery ###

# Country scope: every slave we hold, free or in gangs, is freed and settles in our core states (the capital when we
# own none). The Pens close, contracts, standing orders and auto-renew end, and the slave market closes to us.
GSL_abolish = {
	custom_effect_tooltip = GSL_abolish_tt
	hidden_effect = {
		set_temp_variable = { freed = foe_slaves_var }
		add_to_temp_variable = { freed = GSL_gang_field }
		add_to_temp_variable = { freed = GSL_gang_factory }
		add_to_temp_variable = { freed = GSL_gang_road }
		add_to_temp_variable = { freed = GSL_gang_levy }
		# Rustwreck: everyone serving a Crimson Quota term goes home
		add_to_temp_variable = { freed = RWK_service_total }
		round_temp_variable = freed
		while_loop_effect = {
			limit = {
				check_variable = { freed > 0 }
			}
			if = {
				limit = {
					any_owned_state = { is_core_of = ROOT }
				}
				random_owned_state = {
					limit = { is_core_of = ROOT }
					GSL_settle_population = yes
				}
			}
			else = {
				capital_scope = { GSL_settle_population = yes }
			}
			subtract_from_temp_variable = { freed = 1 }
		}
		set_variable = { foe_slaves_var = 0 }
		set_variable = { GSL_gang_field = 0 }
		set_variable = { GSL_gang_factory = 0 }
		set_variable = { GSL_gang_road = 0 }
		set_variable = { GSL_gang_levy = 0 }
		if = {
			limit = {
				has_dynamic_modifier = { modifier = GSL_the_pens_modifier }
			}
			GSL_update_pens = yes
			remove_dynamic_modifier = { modifier = GSL_the_pens_modifier }
		}
		if = {
			limit = {
				has_dynamic_modifier = { modifier = GSL_market_debt_modifier }
			}
			remove_dynamic_modifier = { modifier = GSL_market_debt_modifier }
		}
		set_variable = { GSL_debt_days = 0 }
		if = {
			limit = {
				has_dynamic_modifier = { modifier = GSL_standing_order_modifier }
			}
			remove_dynamic_modifier = { modifier = GSL_standing_order_modifier }
		}
		set_variable = { GSL_order_left = 0 }
		set_variable = { GSL_order_days = 0 }
		clr_country_flag = GSL_autorenew_flag
		clr_country_flag = GSL_autorenew_paused_flag
		clr_country_flag = slavery_allowed
		if = {
			limit = {
				has_country_flag = RWK_crimson_quotas_flag
			}
			RWK_end_crimson_quotas = yes
		}
		# Red Eye: Fillydelphia stops working slaves to death
		if = {
			limit = {
				has_dynamic_modifier = { modifier = EYE_worked_to_death_modifier }
			}
			remove_dynamic_modifier = { modifier = EYE_worked_to_death_modifier }
		}
	}
}

### Rustwreck: Crimson Quotas (2026-10-15). Citizens drafted on fixed terms (about a year) and lent out by sector;
### RWK_quota_<sector> is the target, RWK_service_<sector> those in service, foe_slaves_var the jungle captives waiting.

# Country scope: opens the Crimson Quotas (The Future in Chains)
RWK_open_crimson_quotas = {
	set_country_flag = RWK_crimson_quotas_flag
	set_variable = { RWK_quota_projects = 0 }
	set_variable = { RWK_service_projects = 0 }
	set_variable = { RWK_quota_army = 0 }
	set_variable = { RWK_service_army = 0 }
	set_variable = { RWK_quota_concessions = 0 }
	set_variable = { RWK_service_concessions = 0 }
	set_variable = { RWK_quota_total = 0 }
	set_variable = { RWK_service_total = 0 }
	set_variable = { RWK_service_cap = 6 }
	set_variable = { RWK_return_acc = 0 }
	add_dynamic_modifier = {
		modifier = RWK_crimson_quotas_modifier
	}
	RWK_update_crimson_quotas = yes
}

# Country scope: recomputes the totals and the Crimson Quotas modifier
RWK_update_crimson_quotas = {
	set_variable = { RWK_service_total = RWK_service_projects }
	add_to_variable = { RWK_service_total = RWK_service_army }
	add_to_variable = { RWK_service_total = RWK_service_concessions }
	set_variable = { RWK_cq_construction = RWK_service_projects }
	multiply_variable = { RWK_cq_construction = 0.03 }
	set_variable = { RWK_cq_supply = RWK_service_army }
	multiply_variable = { RWK_cq_supply = -0.03 }
	set_variable = { RWK_cq_forts = RWK_service_army }
	multiply_variable = { RWK_cq_forts = 0.03 }
	set_variable = { RWK_cq_consumer = RWK_service_concessions }
	multiply_variable = { RWK_cq_consumer = -0.03 }
	set_variable = { RWK_cq_stability = RWK_service_total }
	multiply_variable = { RWK_cq_stability = -0.01 }
	force_update_dynamic_modifier = yes
}

# Country scope: one labourer's term ends (those above their quota first, then the largest sector); they settle as citizens
RWK_end_one_term = {
	if = {
		limit = {
			check_variable = { RWK_service_projects > RWK_quota_projects }
		}
		subtract_from_variable = { RWK_service_projects = 1 }
	}
	else_if = {
		limit = {
			check_variable = { RWK_service_army > RWK_quota_army }
		}
		subtract_from_variable = { RWK_service_army = 1 }
	}
	else_if = {
		limit = {
			check_variable = { RWK_service_concessions > RWK_quota_concessions }
		}
		subtract_from_variable = { RWK_service_concessions = 1 }
	}
	else_if = {
		limit = {
			check_variable = { RWK_service_projects > 0 }
			NOT = { check_variable = { RWK_service_projects < RWK_service_army } }
			NOT = { check_variable = { RWK_service_projects < RWK_service_concessions } }
		}
		subtract_from_variable = { RWK_service_projects = 1 }
	}
	else_if = {
		limit = {
			check_variable = { RWK_service_army > 0 }
			NOT = { check_variable = { RWK_service_army < RWK_service_concessions } }
		}
		subtract_from_variable = { RWK_service_army = 1 }
	}
	else_if = {
		limit = {
			check_variable = { RWK_service_concessions > 0 }
		}
		subtract_from_variable = { RWK_service_concessions = 1 }
	}
	if = {
		limit = {
			any_owned_state = { is_core_of = ROOT }
		}
		random_owned_state = {
			limit = { is_core_of = ROOT }
			GSL_settle_population = yes
		}
	}
	else = {
		capital_scope = { GSL_settle_population = yes }
	}
	subtract_from_variable = { RWK_service_total = 1 }
}

# Country scope: fills the quotas, with jungle captives first, then with citizens drafted from a core state
RWK_crimson_quotas_fill = {
	RWK_update_crimson_quotas = yes
	while_loop_effect = {
		limit = {
			check_variable = { RWK_service_total < RWK_service_cap }
			OR = {
				check_variable = { RWK_service_projects < RWK_quota_projects }
				check_variable = { RWK_service_army < RWK_quota_army }
				check_variable = { RWK_service_concessions < RWK_quota_concessions }
			}
			OR = {
				check_variable = { foe_slaves_var > 0.99 }
				any_owned_state = {
					is_core_of = ROOT
					state_population > 5000
				}
			}
		}
		if = {
			limit = {
				check_variable = { foe_slaves_var > 0.99 }
			}
			subtract_from_variable = { foe_slaves_var = 1 }
		}
		else = {
			random_owned_state = {
				limit = {
					is_core_of = ROOT
					state_population > 5000
				}
				GSL_take_population = yes
			}
		}
		if = {
			limit = {
				check_variable = { RWK_service_projects < RWK_quota_projects }
			}
			add_to_variable = { RWK_service_projects = 1 }
		}
		else_if = {
			limit = {
				check_variable = { RWK_service_army < RWK_quota_army }
			}
			add_to_variable = { RWK_service_army = 1 }
		}
		else_if = {
			limit = {
				check_variable = { RWK_service_concessions < RWK_quota_concessions }
			}
			add_to_variable = { RWK_service_concessions = 1 }
		}
		add_to_variable = { RWK_service_total = 1 }
	}
	RWK_update_crimson_quotas = yes
}

# Country scope, monthly: terms end (a twelfth of those in service, on average), then the gaps are filled
RWK_crimson_quotas_monthly = {
	set_temp_variable = { ending = RWK_service_total }
	divide_temp_variable = { ending = 12 }
	add_to_variable = { RWK_return_acc = ending }
	while_loop_effect = {
		limit = {
			check_variable = { RWK_return_acc > 0.99 }
			check_variable = { RWK_service_total > 0 }
		}
		RWK_end_one_term = yes
		subtract_from_variable = { RWK_return_acc = 1 }
	}
	if = {
		limit = {
			check_variable = { RWK_service_total < 1 }
		}
		set_variable = { RWK_return_acc = 0 }
	}
	RWK_crimson_quotas_fill = yes
}

# Country scope: the Crimson Quotas end (abolition settles those in service, see GSL_abolish)
RWK_end_crimson_quotas = {
	clr_country_flag = RWK_crimson_quotas_flag
	set_variable = { RWK_quota_projects = 0 }
	set_variable = { RWK_service_projects = 0 }
	set_variable = { RWK_quota_army = 0 }
	set_variable = { RWK_service_army = 0 }
	set_variable = { RWK_quota_concessions = 0 }
	set_variable = { RWK_service_concessions = 0 }
	set_variable = { RWK_quota_total = 0 }
	set_variable = { RWK_service_total = 0 }
	if = {
		limit = {
			has_dynamic_modifier = { modifier = RWK_crimson_quotas_modifier }
		}
		remove_dynamic_modifier = { modifier = RWK_crimson_quotas_modifier }
	}
}
