# Rings are bands of that rank (0% = bordering): Rival 10-25%, Archenemy 40-60%, Reserve 65-85%.
FOE_select_antagonists = {
	set_variable = { global.FOE_antagonist_player = ROOT }
	FOE_rank_antagonist_candidates = yes
	set_temp_variable = { FOE_ring = 1 }
	FOE_set_antagonist_band = yes
	FOE_pick_antagonist = yes
	set_temp_variable = { FOE_ring = 2 }
	FOE_set_antagonist_band = yes
	FOE_pick_antagonist = yes
	set_temp_variable = { FOE_ring = 3 }
	FOE_set_antagonist_band = yes
	FOE_pick_antagonist = yes
	FOE_clear_antagonist_ranks = yes
	country_event = { id = foe_antagonists.1 }
}

# Player scope (ROOT). Ranks every candidate by distance and weights it. Used at setup and by refills.
FOE_rank_antagonist_candidates = {
	set_temp_variable = { FOE_pool_size = 0 }
	every_country = {
		limit = { FOE_is_antagonist_candidate = yes }
		set_variable = { FOE_antagonist_rank_pct = 0 }
		add_to_temp_variable = { FOE_pool_size = 1 }
	}

	every_country = {
		limit = { has_variable = FOE_antagonist_rank_pct }
		# Rank = how many candidates are closer to the player than this one.
		set_temp_variable = { FOE_own_distance = closest_distance_to@ROOT }
		every_country = {
			limit = {
				has_variable = FOE_antagonist_rank_pct
				check_variable = { closest_distance_to@ROOT < FOE_own_distance }
			}
			PREV = { add_to_variable = { FOE_antagonist_rank_pct = 1 } }
		}
		multiply_variable = { FOE_antagonist_rank_pct = 100 }
		divide_variable = { FOE_antagonist_rank_pct = FOE_pool_size }

		# Weight = owned states squared
		set_variable = { FOE_antagonist_weight = num_owned_states }
		clamp_variable = {
			var = FOE_antagonist_weight
			max = 10
		}
		multiply_variable = { FOE_antagonist_weight = FOE_antagonist_weight }
		if = {
			limit = { FOE_has_finished_lore_tree = yes }
			multiply_variable = { FOE_antagonist_weight = 3 }
		}

		if = {
			limit = { has_global_flag = FOE_antagonist_selection_audit }
			log = "FOE_ANTAGONIST_AUDIT [THIS.GetTag] rank_pct=[?FOE_antagonist_rank_pct] weight=[?FOE_antagonist_weight] distance=[?FOE_own_distance] pool=[?FOE_pool_size]"
		}
	}
}

# Clears the ranking left by FOE_rank_antagonist_candidates.
FOE_clear_antagonist_ranks = {
	every_country = {
		limit = { has_variable = FOE_antagonist_rank_pct }
		clear_variable = FOE_antagonist_rank_pct
		clear_variable = FOE_antagonist_weight
	}
}

# Temp FOE_ring -> temp FOE_band_low / FOE_band_high, that ring's band of distance rank.
FOE_set_antagonist_band = {
	if = {
		limit = { check_variable = { FOE_ring = 1 } }
		set_temp_variable = { FOE_band_low = 10 }
		set_temp_variable = { FOE_band_high = 25 }
	}
	else_if = {
		limit = { check_variable = { FOE_ring = 2 } }
		set_temp_variable = { FOE_band_low = 40 }
		set_temp_variable = { FOE_band_high = 60 }
	}
	else = {
		set_temp_variable = { FOE_band_low = 65 }
		set_temp_variable = { FOE_band_high = 85 }
	}
}

# Temp FOE_ring
FOE_pick_antagonist = {
	set_temp_variable = { FOE_picked = 0 }

	# Historical rivals first, in 2 of 3 campaigns
	if = {
		limit = {
			any_country = {
				FOE_is_historical_rival = yes
				FOE_is_antagonist_eligible = yes
			}
		}
		random_list = {
			2 = {
				random_country = {
					limit = {
						FOE_is_historical_rival = yes
						FOE_is_antagonist_eligible = yes
					}
					log = "FOE_ANTAGONIST ring=[?FOE_ring] historical rival [THIS.GetTag]"
					FOE_make_antagonist = yes
				}
			}
			1 = {
				log = "FOE_ANTAGONIST ring=[?FOE_ring] historical rival skipped, normal roll"
			}
		}
	}

	set_temp_variable = { FOE_widenings = 0 }
	while_loop_effect = {
		limit = {
			check_variable = { FOE_picked = 0 }
			check_variable = { FOE_widenings < 5 }
		}
		set_temp_variable = { FOE_total_weight = 0 }
		every_country = {
			limit = { FOE_is_antagonist_in_band = yes }
			add_to_temp_variable = { FOE_total_weight = FOE_antagonist_weight }
		}
		if = {
			limit = { check_variable = { FOE_total_weight > 0 } }
			set_temp_variable = { FOE_roll = random }
			multiply_temp_variable = { FOE_roll = FOE_total_weight }
			every_country = {
				limit = { FOE_is_antagonist_in_band = yes }
				if = {
					limit = { check_variable = { FOE_picked = 0 } }
					subtract_from_temp_variable = { FOE_roll = FOE_antagonist_weight }
					if = {
						limit = { check_variable = { FOE_roll < 0 } }
						FOE_make_antagonist = yes
					}
				}
			}
			# Prevent NULL via Rounding
			if = {
				limit = { check_variable = { FOE_picked = 0 } }
				random_country = {
					limit = { FOE_is_antagonist_in_band = yes }
					FOE_make_antagonist = yes
				}
			}
		}
		if = {
			limit = { check_variable = { FOE_picked = 0 } }
			subtract_from_temp_variable = { FOE_band_low = 10 }
			add_to_temp_variable = { FOE_band_high = 10 }
			add_to_temp_variable = { FOE_widenings = 1 }
		}
	}
	if = {
		limit = { check_variable = { FOE_picked = 0 } }
		log = "FOE_ANTAGONIST ring=[?FOE_ring] no candidate found"
	}
}

# Assign Antagonist
FOE_make_antagonist = {
	set_country_flag = FOE_antagonist
	set_variable = { FOE_antagonist_ring = FOE_ring }
	FOE_set_antagonist_formable_states = yes
	FOE_update_antagonist_assessment = yes
	if = {
		limit = { check_variable = { FOE_ring = 1 } }
		set_variable = { global.FOE_antagonist_rival = THIS }
	}
	else_if = {
		limit = { check_variable = { FOE_ring = 2 } }
		set_variable = { global.FOE_antagonist_archenemy = THIS }
	}
	else = {
		set_variable = { global.FOE_antagonist_reserve = THIS }
	}
	set_temp_variable = { FOE_picked = 1 }
	log = "FOE_ANTAGONIST ring=[?FOE_ring] selected=[THIS.GetTag] rank_pct=[?FOE_antagonist_rank_pct] weight=[?FOE_antagonist_weight] band=[?FOE_band_low]-[?FOE_band_high]"
}

# Removes every Antagonist
FOE_clear_antagonists = {
	every_country = {
		limit = { has_country_flag = FOE_antagonist }
		FOE_clear_antagonist_country = yes
	}
	every_state = {
		limit = { has_variable = FOE_antagonist_forts }
		clear_variable = FOE_antagonist_forts
	}
	clear_variable = global.FOE_antagonist_rival
	clear_variable = global.FOE_antagonist_archenemy
	clear_variable = global.FOE_antagonist_reserve
	# Fall phase state
	clr_global_flag = FOE_antagonist_fall_pending_1
	clr_global_flag = FOE_antagonist_player_kill_1
	clr_global_flag = FOE_antagonist_succeeded_1
	clr_global_flag = FOE_antagonist_defeated_1
	clr_global_flag = FOE_antagonist_fallen_slot_1
	clear_variable = global.FOE_antagonist_fallen_1
	clear_variable = global.FOE_antagonist_killer_1
	clr_global_flag = FOE_antagonist_fall_pending_2
	clr_global_flag = FOE_antagonist_player_kill_2
	clr_global_flag = FOE_antagonist_succeeded_2
	clr_global_flag = FOE_antagonist_defeated_2
	clr_global_flag = FOE_antagonist_fallen_slot_2
	clear_variable = global.FOE_antagonist_fallen_2
	clear_variable = global.FOE_antagonist_killer_2
	clr_global_flag = FOE_antagonist_fall_pending_3
	clr_global_flag = FOE_antagonist_player_kill_3
	clr_global_flag = FOE_antagonist_succeeded_3
	clr_global_flag = FOE_antagonist_defeated_3
	clr_global_flag = FOE_antagonist_fallen_slot_3
	clear_variable = global.FOE_antagonist_fallen_3
	clear_variable = global.FOE_antagonist_killer_3
	clear_variable = global.FOE_antagonist_last_fallen
	clear_variable = global.FOE_antagonist_last_ring
}

# Country scope. Removes everything that makes this country an Antagonist (also used when one falls).
FOE_clear_antagonist_country = {
	clr_country_flag = FOE_antagonist
	clr_country_flag = FOE_antagonist_strike_pending
	clear_variable = FOE_antagonist_ring
	clear_array = FOE_antagonist_formable_states
	clear_variable = FOE_antagonist_stall_months
	clear_variable = FOE_antagonist_nudge_months
	clear_variable = FOE_antagonist_wargoal_target
	clear_variable = FOE_antagonist_new_target
	set_variable = { FOE_antagonist_industry_step = 0 }
	set_variable = { FOE_antagonist_research_step = 0 }
	FOE_set_antagonist_industry_idea = yes
	FOE_set_antagonist_research_idea = yes
	clear_variable = FOE_antagonist_industry_step
	clear_variable = FOE_antagonist_research_step
	clear_variable = FOE_antagonist_army_step
	clear_variable = FOE_antagonist_phase
	clear_variable = FOE_antagonist_contact_months
	clear_variable = FOE_antagonist_strike_cooldown
	clear_variable = FOE_antagonist_gl_tier
	clear_variable = FOE_antagonist_status
	clear_variable = FOE_assess_army
	clear_variable = FOE_assess_industry
	clear_variable = FOE_assess_technology
}

FOE_log_antagonists = {
	if = {
		limit = { has_global_flag = FOE_campaign_antagonists_brutal }
		log = "FOE_ANTAGONIST mode=Brutal"
	}
	else_if = {
		limit = { has_global_flag = FOE_campaign_antagonists_enabled }
		log = "FOE_ANTAGONIST mode=Hard"
	}
	else = {
		log = "FOE_ANTAGONIST mode=Off"
	}
	every_country = {
		limit = { has_country_flag = FOE_antagonist }
		log = "FOE_ANTAGONIST [THIS.GetTag] ring=[?FOE_antagonist_ring] industry=[?FOE_antagonist_industry_step] army=[?FOE_antagonist_army_step] research=[?FOE_antagonist_research_step] phase=[?FOE_antagonist_phase]"
	}
}

# Any scope. The player can move to another country mid-game (change_tag_from in lore events: the Commonwealth civil
# war, Bloody Mane -> EYE, ...). Campaign Setup is single-player only, so the one human country is the player.
# Runs at the start of the monthly update and of the hidden fall and refill events.
FOE_update_antagonist_player = {
	if = {
		limit = {
			var:global.FOE_antagonist_player = {
				OR = {
					exists = no
					is_ai = yes
				}
			}
			any_country = { is_ai = no }
		}
		set_temp_variable = { FOE_old_player = global.FOE_antagonist_player }
		random_country = {
			limit = { is_ai = no }
			set_variable = { global.FOE_antagonist_player = THIS }
			log = "FOE_ANTAGONIST the player moved from [?FOE_old_player.GetTag] to [THIS.GetTag]"
			# Hidden fall and refill events still queued on the old country do nothing there; queue them again here
			if = {
				limit = {
					OR = {
						has_global_flag = FOE_antagonist_fall_pending_1
						has_global_flag = FOE_antagonist_fall_pending_2
						has_global_flag = FOE_antagonist_fall_pending_3
					}
				}
				country_event = { id = foe_antagonists.8 days = 1 }
			}
			if = {
				limit = { has_global_flag = FOE_campaign_antagonists_brutal }
				if = {
					limit = { has_global_flag = FOE_antagonist_fallen_slot_1 }
					country_event = { id = foe_antagonists.10 days = 90 }
				}
				if = {
					limit = { has_global_flag = FOE_antagonist_fallen_slot_2 }
					country_event = { id = foe_antagonists.11 days = 90 }
				}
				if = {
					limit = { has_global_flag = FOE_antagonist_fallen_slot_3 }
					country_event = { id = foe_antagonists.12 days = 90 }
				}
			}
			else = {
				if = {
					limit = { has_global_flag = FOE_antagonist_fallen_slot_1 }
					country_event = { id = foe_antagonists.10 days = 180 }
				}
				if = {
					limit = { has_global_flag = FOE_antagonist_fallen_slot_2 }
					country_event = { id = foe_antagonists.11 days = 180 }
				}
				if = {
					limit = { has_global_flag = FOE_antagonist_fallen_slot_3 }
					country_event = { id = foe_antagonists.12 days = 180 }
				}
			}
			# The player took over an Antagonist: its slot falls as if the old country had taken it, and is refilled
			if = {
				limit = { has_country_flag = FOE_antagonist }
				set_variable = { FOE_antagonist_killer = FOE_old_player }
				FOE_antagonist_falls = yes
			}
		}
	}
}

# Country scope, an Antagonist. Monthly. In the player's faction it could never strike and its system help would go to
# the player's side, so it is put out of the faction again.
FOE_keep_antagonist_out_of_player_faction = {
	if = {
		limit = { is_in_faction_with = var:global.FOE_antagonist_player }
		if = {
			limit = { is_faction_leader = yes }
			var:global.FOE_antagonist_player = { leave_faction = yes }
		}
		else = {
			leave_faction = yes
		}
		log = "FOE_ANTAGONIST [THIS.GetTag] and the player [?global.FOE_antagonist_player.GetTag] no longer share a faction"
	}
}

# Determine Countries that the Antagonist should target; Runs Monthly
FOE_update_antagonist_expansion = {
	if = {
		limit = {
			has_war = no
			is_subject = no
			# Paused while a strike countdown runs
			NOT = { has_country_flag = FOE_antagonist_strike_pending }
		}
		set_temp_variable = { FOE_nudge_wait = 2 }
		set_temp_variable = { FOE_stall_wait = 6 }
		set_temp_variable = { FOE_factory_target = 0.9 }
		if = {
			limit = { has_global_flag = FOE_campaign_antagonists_brutal }
			set_temp_variable = { FOE_nudge_wait = 1 }
			set_temp_variable = { FOE_stall_wait = 3 }
			set_temp_variable = { FOE_factory_target = 1.2 }
		}
		# The system only pushes an Antagonist that lags behind the player
		set_temp_variable = { FOE_own_factories = num_of_factories }
		var:global.FOE_antagonist_player = {
			set_temp_variable = { FOE_player_factories = num_of_factories }
		}
		multiply_temp_variable = { FOE_player_factories = FOE_factory_target }
		set_temp_variable = { FOE_lagging = 0 }
		if = {
			limit = { check_variable = { FOE_own_factories < FOE_player_factories } }
			set_temp_variable = { FOE_lagging = 1 }
		}
		FOE_scan_antagonist_targets = yes

		if = {
			limit = { has_variable = FOE_antagonist_wargoal_target }
			set_variable = { FOE_antagonist_stall_months = 0 }
			add_to_variable = { FOE_antagonist_nudge_months = 1 }
			if = {
				limit = {
					NOT = { check_variable = { FOE_antagonist_nudge_months < FOE_nudge_wait } }
					check_variable = { FOE_lagging = 1 }
				}
				set_variable = { FOE_antagonist_nudge_months = 0 }
				var:FOE_antagonist_wargoal_target = {
					if = {
						limit = { has_non_aggression_pact_with = ROOT }
						diplomatic_relation = {
							country = ROOT
							relation = non_aggression_pact
							active = no
						}
					}
					ROOT = {
						declare_war_on = {
							target = PREV
							type = annex_everything
						}
					}
					if = {
						limit = { has_war_with = ROOT }
						log = "FOE_ANTAGONIST_EXPANSION [ROOT.GetTag] declares war on [THIS.GetTag]"
					}
					else = {
						log = "FOE_ANTAGONIST_EXPANSION [ROOT.GetTag] declared on [THIS.GetTag] but is not at war with it yet"
					}
				}
			}
		}
		else = {
			set_variable = { FOE_antagonist_nudge_months = 0 }
			add_to_variable = { FOE_antagonist_stall_months = 1 }
			if = {
				limit = {
					NOT = { check_variable = { FOE_antagonist_stall_months < FOE_stall_wait } }
					check_variable = { FOE_lagging = 1 }
				}
				if = {
					limit = { has_variable = FOE_antagonist_new_target }
					set_variable = { FOE_antagonist_stall_months = 0 }
					var:FOE_antagonist_new_target = {
						ROOT = {
							create_wargoal = {
								type = annex_everything
								target = PREV
							}
						}
						log = "FOE_ANTAGONIST_EXPANSION [ROOT.GetTag] stalled: war goal on [THIS.GetTag] (formable=[?FOE_best_new_formable] divisions=[?FOE_best_new_divisions])"
					}
				}
				else_if = {
					limit = { check_variable = { FOE_antagonist_stall_months = FOE_stall_wait } }
					log = "FOE_ANTAGONIST_EXPANSION [ROOT.GetTag] stalled: no beatable neighbour in its region, retrying monthly"
				}
			}
		}
		# Logged once, when a timer runs out while it is ahead.
		if = {
			limit = {
				check_variable = { FOE_lagging = 0 }
				OR = {
					check_variable = { FOE_antagonist_stall_months = FOE_stall_wait }
					check_variable = { FOE_antagonist_nudge_months = FOE_nudge_wait }
				}
			}
			log = "FOE_ANTAGONIST_EXPANSION [ROOT.GetTag] is ahead of the player ([?FOE_own_factories] factories, help stops at [?FOE_player_factories]): system help paused"
		}
	}
	else = {
		set_variable = { FOE_antagonist_stall_months = 0 }
		set_variable = { FOE_antagonist_nudge_months = 0 }
	}
}

# Check elligible expansion routes
FOE_scan_antagonist_targets = {
	clear_variable = FOE_antagonist_wargoal_target
	clear_variable = FOE_antagonist_new_target
	set_temp_variable = { FOE_best_wargoal_divisions = -1 }
	set_temp_variable = { FOE_best_new_divisions = -1 }
	set_temp_variable = { FOE_best_new_formable = 0 }
	every_neighbor_country = {
		limit = {
			is_ai = yes
			is_subject = no
			is_in_faction = no
			NOT = { has_country_flag = FOE_antagonist }
			NOT = { has_truce_with = ROOT }
			NOT = { any_country = { has_guaranteed = PREV } }
			strength_ratio = {
				tag = ROOT
				ratio < 0.67
			}
		}
		# Don't step on other antagonist's toes
		set_temp_variable = { FOE_distance_to_us = closest_distance_to@ROOT }
		set_temp_variable = { FOE_in_region = 1 }
		var:global.FOE_antagonist_player = {
			if = {
				limit = { NOT = { check_variable = { closest_distance_to@PREV > FOE_distance_to_us } } }
				set_temp_variable = { FOE_in_region = 0 }
			}
		}
		every_country = {
			limit = {
				has_country_flag = FOE_antagonist
				NOT = { tag = ROOT }
				NOT = { check_variable = { closest_distance_to@PREV > FOE_distance_to_us } }
			}
			set_temp_variable = { FOE_in_region = 0 }
		}

		if = {
			limit = { check_variable = { FOE_in_region = 1 } }
			set_temp_variable = { FOE_divisions = num_divisions }
			if = {
				limit = {
					ROOT = { has_wargoal_against = PREV }
					OR = {
						check_variable = { FOE_best_wargoal_divisions < 0 }
						check_variable = { FOE_divisions < FOE_best_wargoal_divisions }
					}
				}
				ROOT = { set_variable = { FOE_antagonist_wargoal_target = PREV } }
				set_temp_variable = { FOE_best_wargoal_divisions = FOE_divisions }
			}
			set_temp_variable = { FOE_formable = 0 }
			if = {
				limit = {
					any_of_scopes = {
						array = ROOT.FOE_antagonist_formable_states
						is_owned_by = PREV
					}
				}
				set_temp_variable = { FOE_formable = 1 }
			}
			if = {
				limit = {
					OR = {
						check_variable = { FOE_best_new_divisions < 0 }
						check_variable = { FOE_formable > FOE_best_new_formable }
						AND = {
							check_variable = { FOE_formable = FOE_best_new_formable }
							check_variable = { FOE_divisions < FOE_best_new_divisions }
						}
					}
				}
				ROOT = { set_variable = { FOE_antagonist_new_target = PREV } }
				set_temp_variable = { FOE_best_new_divisions = FOE_divisions }
				set_temp_variable = { FOE_best_new_formable = FOE_formable }
			}
		}
	}
}


# Rubber Band Effects - Let's the Antagonist keep up with a player that is outscaling them
FOE_update_antagonist_rubber_band = {
	set_temp_variable = { FOE_share = 0.9 }
	set_temp_variable = { FOE_research_share = 1.1 }
	if = {
		limit = { has_global_flag = FOE_campaign_antagonists_brutal }
		set_temp_variable = { FOE_share = 1.2 }
		set_temp_variable = { FOE_research_share = 1.4 }
	}
	var:global.FOE_antagonist_player = {
		set_temp_variable = { FOE_player_factories = num_of_factories }
		set_temp_variable = { FOE_player_divisions = num_divisions }
		set_temp_variable = { FOE_player_techs = num_researched_technologies }
	}

	# Industry.
	set_temp_variable = { FOE_band_own = num_of_factories }
	set_temp_variable = { FOE_band_target = FOE_player_factories }
	multiply_temp_variable = { FOE_band_target = FOE_share }
	FOE_antagonist_band_step = yes
	if = {
		limit = { NOT = { check_variable = { FOE_antagonist_industry_step = FOE_band_step } } }
		set_variable = { FOE_antagonist_industry_step = FOE_band_step }
		FOE_set_antagonist_industry_idea = yes
		log = "FOE_ANTAGONIST_BAND [ROOT.GetTag] industry step [?FOE_antagonist_industry_step] ([?FOE_band_own] factories, target [?FOE_band_target])"
	}

	# Military
	set_temp_variable = { FOE_band_own = num_divisions }
	set_temp_variable = { FOE_band_target = FOE_player_divisions }
	multiply_temp_variable = { FOE_band_target = FOE_share }
	FOE_antagonist_band_step = yes
	set_variable = { FOE_antagonist_army_step = FOE_band_step }
	if = {
		limit = { check_variable = { FOE_antagonist_army_step > 0 } }
		set_temp_variable = { FOE_army_drip = FOE_antagonist_army_step }
		multiply_temp_variable = { FOE_army_drip = 50 }
		add_manpower = FOE_army_drip
		FOE_give_antagonist_weapons = yes
		# Army XP, so the AI can afford its target templates: 5 per step
		for_loop_effect = {
			end = FOE_antagonist_army_step
			army_experience = 5
		}
	}

	# Research
	set_temp_variable = { FOE_band_own = num_researched_technologies }
	set_temp_variable = { FOE_band_target = FOE_player_techs }
	multiply_temp_variable = { FOE_band_target = FOE_research_share }
	FOE_antagonist_band_step = yes
	if = {
		limit = { NOT = { check_variable = { FOE_antagonist_research_step = FOE_band_step } } }
		set_variable = { FOE_antagonist_research_step = FOE_band_step }
		FOE_set_antagonist_research_idea = yes
		log = "FOE_ANTAGONIST_BAND [ROOT.GetTag] research step [?FOE_antagonist_research_step] ([?FOE_band_own] techs, target [?FOE_band_target])"
	}

	if = {
		limit = { has_global_flag = FOE_antagonist_band_audit }
		log = "FOE_ANTAGONIST_BAND [ROOT.GetTag] industry=[?FOE_antagonist_industry_step] army=[?FOE_antagonist_army_step] research=[?FOE_antagonist_research_step]"
	}
}

# Determine how much of a boost the Antagonist gets
FOE_antagonist_band_step = {
	set_temp_variable = { FOE_band_step = 0 }
	if = {
		limit = { check_variable = { FOE_band_target > 0 } }
		set_temp_variable = { FOE_band_ratio = FOE_band_own }
		divide_temp_variable = { FOE_band_ratio = FOE_band_target }
		if = {
			limit = { check_variable = { FOE_band_ratio < 0.3 } }
			set_temp_variable = { FOE_band_step = 5 }
		}
		else_if = {
			limit = { check_variable = { FOE_band_ratio < 0.45 } }
			set_temp_variable = { FOE_band_step = 4 }
		}
		else_if = {
			limit = { check_variable = { FOE_band_ratio < 0.6 } }
			set_temp_variable = { FOE_band_step = 3 }
		}
		else_if = {
			limit = { check_variable = { FOE_band_ratio < 0.75 } }
			set_temp_variable = { FOE_band_step = 2 }
		}
		else_if = {
			limit = { check_variable = { FOE_band_ratio < 0.9 } }
			set_temp_variable = { FOE_band_step = 1 }
		}
	}
}

# Script to give most recent Antagonist weapon
FOE_give_antagonist_weapons = {
	set_temp_variable = { FOE_rifles = FOE_army_drip }
	set_temp_variable = { FOE_energy = 0 }
	set_temp_variable = { FOE_energy_need = num_target_equipment_in_armies@mew_equipment }
	if = {
		limit = { check_variable = { FOE_energy_need > 0 } }
		#Give Energy weapons if they are being used
		set_temp_variable = { FOE_weapon_need = num_target_equipment_in_armies@infantry_equipment }
		add_to_temp_variable = { FOE_weapon_need = FOE_energy_need }
		set_temp_variable = { FOE_energy = FOE_energy_need }
		divide_temp_variable = { FOE_energy = FOE_weapon_need }
		multiply_temp_variable = { FOE_energy = FOE_army_drip }
		round_temp_variable = FOE_energy
		subtract_from_temp_variable = { FOE_rifles = FOE_energy }
	}
	if = {
		limit = { check_variable = { FOE_rifles > 0 } }
		if = {
			limit = { has_tech = ballistic_weaponry_tech_4 }
			add_equipment_to_stockpile = { type = ballistic_equipment_4 amount = FOE_rifles producer = OWE }
		}
		else_if = {
			limit = { has_tech = ballistic_weaponry_tech_3 }
			add_equipment_to_stockpile = { type = ballistic_equipment_3 amount = FOE_rifles producer = OWE }
		}
		else_if = {
			limit = { has_tech = ballistic_weaponry_tech_2 }
			add_equipment_to_stockpile = { type = ballistic_equipment_2 amount = FOE_rifles producer = OWE }
		}
		else_if = {
			limit = { has_tech = ballistic_weaponry_tech_1 }
			add_equipment_to_stockpile = { type = ballistic_equipment_1 amount = FOE_rifles producer = OWE }
		}
		else = {
			add_equipment_to_stockpile = { type = infantry_equipment_1 amount = FOE_rifles producer = OWE }
		}
	}
	if = {
		limit = { check_variable = { FOE_energy > 0 } }
		if = {
			limit = { has_tech = energy_weaponry_tech_4 }
			add_equipment_to_stockpile = { type = energy_equipment_4 amount = FOE_energy producer = OWE }
		}
		else_if = {
			limit = { has_tech = energy_weaponry_tech_3 }
			add_equipment_to_stockpile = { type = energy_equipment_3 amount = FOE_energy producer = OWE }
		}
		else_if = {
			limit = { has_tech = energy_weaponry_tech_2 }
			add_equipment_to_stockpile = { type = energy_equipment_2 amount = FOE_energy producer = OWE }
		}
		else = {
			add_equipment_to_stockpile = { type = energy_equipment_1 amount = FOE_energy producer = OWE }
		}
	}
	if = {
		limit = { has_global_flag = FOE_antagonist_band_audit }
		log = "FOE_ANTAGONIST_BAND [ROOT.GetTag] weapons: [?FOE_rifles] rifles, [?FOE_energy] energy weapons"
	}
}

#Determine if Confrontation can happen this month
FOE_update_antagonist_confrontation = {
	set_temp_variable = { FOE_contact_wait = 6 }
	set_temp_variable = { FOE_strike_cooldown = 12 }
	set_temp_variable = { FOE_fort_cap = 2 }
	if = {
		limit = { has_global_flag = FOE_campaign_antagonists_brutal }
		set_temp_variable = { FOE_contact_wait = 3 }
		set_temp_variable = { FOE_strike_cooldown = 6 }
		set_temp_variable = { FOE_fort_cap = 4 }
	}

	# Phases
	if = {
		limit = { check_variable = { FOE_antagonist_phase > 0 } }
		add_to_variable = { FOE_antagonist_contact_months = 1 }
	}
	else_if = {
		limit = {
			var:global.FOE_antagonist_player = {
				OR = {
					is_neighbor_of = ROOT
					has_war_with = ROOT
				}
			}
		}
		FOE_antagonist_contact = yes
	}
	if = {
		limit = {
			check_variable = { FOE_antagonist_phase < 2 }
			has_war_with = var:global.FOE_antagonist_player
		}
		set_variable = { FOE_antagonist_phase = 2 }
		log = "FOE_ANTAGONIST_CONFRONTATION [ROOT.GetTag] Confrontation: at war with the player"
		# The Archenemy ranks up at Confrontation
		if = {
			limit = { check_variable = { FOE_antagonist_ring = 2 } }
			FOE_antagonist_leader_rank_up = yes
		}
	}

	# Cooldown
	if = {
		limit = { has_war_with = var:global.FOE_antagonist_player }
		set_variable = { FOE_antagonist_strike_cooldown = FOE_strike_cooldown }
	}
	else_if = {
		limit = { check_variable = { FOE_antagonist_strike_cooldown > 0 } }
		subtract_from_variable = { FOE_antagonist_strike_cooldown = 1 }
	}

	# Forts
	if = {
		limit = {
			check_variable = { FOE_antagonist_phase > 0 }
			NOT = { has_war_with = var:global.FOE_antagonist_player }
		}
		every_owned_state = {
			limit = {
				check_variable = { FOE_antagonist_forts < FOE_fort_cap }
				any_neighbor_state = { is_owned_by = var:global.FOE_antagonist_player }
			}
			add_building_construction = {
				type = bunker
				level = 1
				instant_build = yes
				province = {
					all_provinces = yes
					limit_to_border = yes
				}
			}
			add_to_variable = { FOE_antagonist_forts = 1 }
		}
	}

	#Begin First strike, preferably forcing 2-front war
	if = {
		limit = {
			check_variable = { FOE_antagonist_phase > 0 }
			NOT = { check_variable = { FOE_antagonist_contact_months < FOE_contact_wait } }
			NOT = { check_variable = { FOE_antagonist_strike_cooldown > 0 } }
			has_war = no
			is_subject = no
			NOT = { has_country_flag = FOE_antagonist_strike_pending }
			var:global.FOE_antagonist_player = {
				is_neighbor_of = ROOT
				NOT = { is_in_faction_with = ROOT }
			}
		}
		set_temp_variable = { FOE_strike = 0 }
		if = {
			limit = { has_global_flag = FOE_campaign_antagonists_brutal }
			if = {
				limit = {
					var:global.FOE_antagonist_player = {
						OR = {
							strength_ratio = {
								tag = ROOT
								ratio < 0.84
							}
							AND = {
								has_war = yes
								strength_ratio = {
									tag = ROOT
									ratio < 2.5
								}
							}
						}
					}
				}
				set_temp_variable = { FOE_strike = 1 }
			}
		}
		else_if = {
			limit = {
				var:global.FOE_antagonist_player = {
					#Antagonists never pile on in Hard.
					any_enemy_country = {
						NOT = { has_country_flag = FOE_antagonist }
					}
					strength_ratio = {
						tag = ROOT
						ratio < 1.67
					}
				}
			}
			set_temp_variable = { FOE_strike = 1 }
		}
		if = {
			limit = { check_variable = { FOE_strike = 1 } }
			FOE_start_antagonist_strike = yes
		}
	}

	# Tracker status: 1 Consolidating, 2 On our border, 3 Ready to strike (grace and cooldown over), 4 At war with us,
	# 5 Preparing to attack us (strike countdown running)
	set_variable = { FOE_antagonist_status = 1 }
	if = {
		limit = { has_war_with = var:global.FOE_antagonist_player }
		set_variable = { FOE_antagonist_status = 4 }
	}
	else_if = {
		limit = { has_country_flag = FOE_antagonist_strike_pending }
		set_variable = { FOE_antagonist_status = 5 }
	}
	else_if = {
		limit = {
			check_variable = { FOE_antagonist_phase > 0 }
			NOT = { check_variable = { FOE_antagonist_contact_months < FOE_contact_wait } }
			NOT = { check_variable = { FOE_antagonist_strike_cooldown > 0 } }
		}
		set_variable = { FOE_antagonist_status = 3 }
	}
	else_if = {
		limit = { check_variable = { FOE_antagonist_phase > 0 } }
		set_variable = { FOE_antagonist_status = 2 }
	}

	if = {
		limit = { has_global_flag = FOE_antagonist_band_audit }
		log = "FOE_ANTAGONIST_CONFRONTATION [ROOT.GetTag] phase=[?FOE_antagonist_phase] contact_months=[?FOE_antagonist_contact_months] cooldown=[?FOE_antagonist_strike_cooldown]"
	}
}

# Fight, begin
FOE_antagonist_contact = {
	set_variable = { FOE_antagonist_phase = 1 }
	set_variable = { FOE_antagonist_contact_months = 0 }
	# The Rival and the Distant Threat rank up at Contact
	if = {
		limit = { NOT = { check_variable = { FOE_antagonist_ring = 2 } } }
		FOE_antagonist_leader_rank_up = yes
	}
	var:global.FOE_antagonist_player = {
		country_event = { id = foe_antagonists.4 }
	}
	log = "FOE_ANTAGONIST_CONFRONTATION [ROOT.GetTag] Contact: borders the player"
}

# Country scope (ROOT), an Antagonist whose strike check passed. Starts the strike countdown: a targeted mission on the
# player that declares the war when it times out, 30 days on Hard / 10 on Brutal (common/decisions/foe_antagonist_decisions.txt).
FOE_start_antagonist_strike = {
	set_country_flag = FOE_antagonist_strike_pending
	var:global.FOE_antagonist_player = {
		ROOT = {
			if = {
				limit = { has_global_flag = FOE_campaign_antagonists_brutal }
				activate_targeted_decision = { target = PREV decision = FOE_antagonist_strike_mission_brutal }
			}
			else = {
				activate_targeted_decision = { target = PREV decision = FOE_antagonist_strike_mission }
			}
		}
		country_event = { id = foe_antagonists.13 }
		log = "FOE_ANTAGONIST_CONFRONTATION [ROOT.GetTag] starts its strike countdown on the player [THIS.GetTag]"
	}
}

#Declare war on player
FOE_antagonist_first_strike = {
	var:global.FOE_antagonist_player = {
		if = {
			limit = { has_non_aggression_pact_with = ROOT }
			diplomatic_relation = {
				country = ROOT
				relation = non_aggression_pact
				active = no
			}
		}
		ROOT = {
			declare_war_on = {
				target = PREV
				type = annex_everything
			}
		}
		if = {
			limit = { has_war_with = ROOT }
			country_event = { id = foe_antagonists.5 }
			log = "FOE_ANTAGONIST_CONFRONTATION [ROOT.GetTag] first strike on the player [THIS.GetTag]"
		}
		else = {
			log = "FOE_ANTAGONIST_CONFRONTATION [ROOT.GetTag] declared on the player [THIS.GetTag] but is not at war with them yet"
		}
	}
}

# Country scope, an Antagonist. Monthly. Compares its divisions, factories and researched techs with the player's
# and stores a band 1-7 for each: Far weaker < 50% / Weaker < 75% / Slightly weaker < 90% / Equal up to 110% /
# Slightly stronger < 133% / Stronger < 200% / Far stronger.
FOE_update_antagonist_assessment = {
	var:global.FOE_antagonist_player = {
		set_temp_variable = { FOE_player_divisions = num_divisions }
		set_temp_variable = { FOE_player_factories = num_of_factories }
		set_temp_variable = { FOE_player_techs = num_researched_technologies }
	}

	set_temp_variable = { FOE_assess_own = num_divisions }
	set_temp_variable = { FOE_assess_player = FOE_player_divisions }
	FOE_antagonist_assessment_band = yes
	set_variable = { FOE_assess_army = FOE_assess_band }

	set_temp_variable = { FOE_assess_own = num_of_factories }
	set_temp_variable = { FOE_assess_player = FOE_player_factories }
	FOE_antagonist_assessment_band = yes
	set_variable = { FOE_assess_industry = FOE_assess_band }

	set_temp_variable = { FOE_assess_own = num_researched_technologies }
	set_temp_variable = { FOE_assess_player = FOE_player_techs }
	FOE_antagonist_assessment_band = yes
	set_variable = { FOE_assess_technology = FOE_assess_band }
}

# Temp FOE_assess_own / FOE_assess_player -> temp FOE_assess_band 1-7. A player with none counts as Far weaker
# than any Antagonist with some, and Equal to one with none.
FOE_antagonist_assessment_band = {
	set_temp_variable = { FOE_assess_band = 4 }
	if = {
		limit = { check_variable = { FOE_assess_player > 0 } }
		set_temp_variable = { FOE_assess_ratio = FOE_assess_own }
		divide_temp_variable = { FOE_assess_ratio = FOE_assess_player }
		if = {
			limit = { check_variable = { FOE_assess_ratio < 0.5 } }
			set_temp_variable = { FOE_assess_band = 1 }
		}
		else_if = {
			limit = { check_variable = { FOE_assess_ratio < 0.75 } }
			set_temp_variable = { FOE_assess_band = 2 }
		}
		else_if = {
			limit = { check_variable = { FOE_assess_ratio < 0.9 } }
			set_temp_variable = { FOE_assess_band = 3 }
		}
		else_if = {
			limit = { check_variable = { FOE_assess_ratio > 2 } }
			set_temp_variable = { FOE_assess_band = 7 }
		}
		else_if = {
			limit = { check_variable = { FOE_assess_ratio > 1.33 } }
			set_temp_variable = { FOE_assess_band = 6 }
		}
		else_if = {
			limit = { check_variable = { FOE_assess_ratio > 1.1 } }
			set_temp_variable = { FOE_assess_band = 5 }
		}
	}
	else_if = {
		limit = { check_variable = { FOE_assess_own > 0 } }
		set_temp_variable = { FOE_assess_band = 7 }
	}
}

# Fall: an Antagonist is annexed or puppeted. The player's kill leaves the slot empty ("Defeated by us"); any other
# kill refills the slot after 6 months (Hard) / 3 (Brutal). A civil-war winner takes the slot over instead.
# Per-ring state: global flags/vars ending in _1 Rival / _2 Archenemy / _3 Distant Threat.
###########################################################################

# Country scope, the fallen Antagonist, with FOE_antagonist_killer set. From on_annex, on_puppet, on_civil_war_end.
# The outcome is settled a day later (foe_antagonists.8), so a civil war can claim the slot first.
FOE_antagonist_falls = {
	set_variable = { FOE_antagonist_fallen_ring = FOE_antagonist_ring }
	if = {
		limit = { has_variable = FOE_antagonist_gl_tier }
		set_variable = { FOE_antagonist_fallen_gl_tier = FOE_antagonist_gl_tier }
	}
	# The player's side: the player, its subjects, its faction, or anyone while we were at war with the player
	set_temp_variable = { FOE_player_kill = 0 }
	if = {
		limit = {
			OR = {
				has_war_with = var:global.FOE_antagonist_player
				check_variable = { FOE_antagonist_status = 4 }
				var:FOE_antagonist_killer = {
					OR = {
						tag = var:global.FOE_antagonist_player
						is_subject_of = var:global.FOE_antagonist_player
						is_in_faction_with = var:global.FOE_antagonist_player
					}
				}
			}
		}
		set_temp_variable = { FOE_player_kill = 1 }
	}
	if = {
		limit = { check_variable = { FOE_antagonist_ring = 1 } }
		set_global_flag = FOE_antagonist_fall_pending_1
		set_variable = { global.FOE_antagonist_fallen_1 = THIS }
		set_variable = { global.FOE_antagonist_killer_1 = FOE_antagonist_killer }
		if = {
			limit = { check_variable = { FOE_player_kill = 1 } }
			set_global_flag = FOE_antagonist_player_kill_1
		}
	}
	else_if = {
		limit = { check_variable = { FOE_antagonist_ring = 2 } }
		set_global_flag = FOE_antagonist_fall_pending_2
		set_variable = { global.FOE_antagonist_fallen_2 = THIS }
		set_variable = { global.FOE_antagonist_killer_2 = FOE_antagonist_killer }
		if = {
			limit = { check_variable = { FOE_player_kill = 1 } }
			set_global_flag = FOE_antagonist_player_kill_2
		}
	}
	else_if = {
		limit = { check_variable = { FOE_antagonist_ring = 3 } }
		set_global_flag = FOE_antagonist_fall_pending_3
		set_variable = { global.FOE_antagonist_fallen_3 = THIS }
		set_variable = { global.FOE_antagonist_killer_3 = FOE_antagonist_killer }
		if = {
			limit = { check_variable = { FOE_player_kill = 1 } }
			set_global_flag = FOE_antagonist_player_kill_3
		}
	}
	log = "FOE_ANTAGONIST_FALL [THIS.GetTag] ring=[?FOE_antagonist_ring] fell to [?FOE_antagonist_killer.GetTag] player_kill=[?FOE_player_kill]"
	FOE_clear_antagonist_country = yes
	var:global.FOE_antagonist_player = {
		country_event = { id = foe_antagonists.8 days = 1 }
	}
}

# Country scope, the winner of a civil war (ROOT); FROM = the side that was annexed. If FROM was an Antagonist, the
# winner takes its slot and keeps its Great Leader tier; no refill.
FOE_antagonist_civil_war_successor = {
	if = {
		limit = { FROM = { has_country_flag = FOE_antagonist } }
		FROM = {
			set_variable = { FOE_antagonist_killer = ROOT }
			FOE_antagonist_falls = yes
		}
	}
	set_temp_variable = { FOE_ring = FROM.FOE_antagonist_fallen_ring }
	if = {
		limit = { check_variable = { FOE_ring = 1 } }
		set_global_flag = FOE_antagonist_succeeded_1
	}
	else_if = {
		limit = { check_variable = { FOE_ring = 2 } }
		set_global_flag = FOE_antagonist_succeeded_2
	}
	else_if = {
		limit = { check_variable = { FOE_ring = 3 } }
		set_global_flag = FOE_antagonist_succeeded_3
	}
	FOE_make_antagonist = yes
	if = {
		limit = { FROM = { has_variable = FOE_antagonist_fallen_gl_tier } }
		set_variable = { FOE_antagonist_gl_tier = FROM.FOE_antagonist_fallen_gl_tier }
	}
	log = "FOE_ANTAGONIST_FALL [THIS.GetTag] won the civil war and takes ring [?FOE_ring]"
}

# Country scope, a fallen Antagonist, a day after its fall. on_annex and on_civil_war_end, which read these, have fired

# Player scope (ROOT), temp FOE_ring; from foe_antagonists.10 / .11 / .12. Picks a new Antagonist for that ring the
# same way as at setup; the nation that killed the last one can't take the slot. No candidate: try again later.
FOE_refill_antagonist_slot = {
	if = {
		limit = {
			check_variable = { FOE_ring = 1 }
			has_variable = global.FOE_antagonist_killer_1
		}
		var:global.FOE_antagonist_killer_1 = { set_country_flag = FOE_antagonist_excluded }
	}
	else_if = {
		limit = {
			check_variable = { FOE_ring = 2 }
			has_variable = global.FOE_antagonist_killer_2
		}
		var:global.FOE_antagonist_killer_2 = { set_country_flag = FOE_antagonist_excluded }
	}
	else_if = {
		limit = {
			check_variable = { FOE_ring = 3 }
			has_variable = global.FOE_antagonist_killer_3
		}
		var:global.FOE_antagonist_killer_3 = { set_country_flag = FOE_antagonist_excluded }
	}
	FOE_rank_antagonist_candidates = yes
	FOE_set_antagonist_band = yes
	FOE_pick_antagonist = yes
	FOE_clear_antagonist_ranks = yes
	every_country = {
		limit = { has_country_flag = FOE_antagonist_excluded }
		clr_country_flag = FOE_antagonist_excluded
	}
	if = {
		limit = { check_variable = { FOE_picked = 1 } }
		set_variable = { global.FOE_antagonist_last_ring = FOE_ring }
		if = {
			limit = { check_variable = { FOE_ring = 1 } }
			clr_global_flag = FOE_antagonist_fallen_slot_1
			set_variable = { global.FOE_antagonist_last_fallen = global.FOE_antagonist_fallen_1 }
			var:global.FOE_antagonist_rival = { FOE_set_refill_antagonist_leader = yes }
		}
		else_if = {
			limit = { check_variable = { FOE_ring = 2 } }

# Country scope, a replacement Antagonist, temp FOE_ring. A Great Leader keeps class and subclass and is raised to the
# ring's starting tier; a leader who is none gets an Antagonist Great Leader.
FOE_set_refill_antagonist_leader = {
	if = {
		limit = {
			has_global_flag = FOE_campaign_great_leaders_enabled
			FOE_great_leader_country_valid = yes
		}
		if = {
			limit = { var:country_leader = { has_character_flag = FOE_great_leader_character } }
			set_variable = { FOE_antagonist_gl_tier = 1 }
			if = {
				limit = {
					OR = {
						FOE_has_legendary_or_mythic_conqueror_tier = yes
						FOE_has_legendary_or_mythic_administrator_tier = yes
						FOE_has_legendary_or_mythic_industrialist_tier = yes
					}
				}
				set_variable = { FOE_antagonist_gl_tier = 2 }
			}
			if = {
				limit = {
					OR = {
						FOE_has_mythic_conqueror_tier = yes
						FOE_has_mythic_administrator_tier = yes
						FOE_has_mythic_industrialist_tier = yes
					}
				}
				set_variable = { FOE_antagonist_gl_tier = 3 }
			}
			set_temp_variable = { FOE_start_tier = 1 }
			if = {
				limit = { check_variable = { FOE_ring = 2 } }
				set_temp_variable = { FOE_start_tier = 2 }
			}
			if = {
				limit = { has_global_flag = FOE_campaign_antagonists_brutal }
				add_to_temp_variable = { FOE_start_tier = 1 }
			}
			if = {
				limit = { check_variable = { FOE_antagonist_gl_tier < FOE_start_tier } }
				FOE_antagonist_leader_rank_up = yes
			}
			if = {
				limit = { check_variable = { FOE_antagonist_gl_tier < FOE_start_tier } }
				FOE_antagonist_leader_rank_up = yes
			}
		}
		else = {
			FOE_assign_antagonist_great_leader = yes
		}
	}
}
