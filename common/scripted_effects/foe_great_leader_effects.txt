#Nomadic Khan Subclass
FOE_initialize_nomadic_khan = {
	add_country_leader_trait = FOE_nomadic_khan_trait
	var:country_leader = {
		set_character_flag = FOE_great_leader_character
		set_character_flag = FOE_nomadic_khan_character
	}
	if = {
		limit = { NOT = { has_variable = FOE_nomadic_court_state } }
		capital_scope = {
			ROOT = { set_variable = { FOE_nomadic_court_state = PREV.id } }
			add_dynamic_modifier = { modifier = FOE_nomadic_court_modifier scope = ROOT }
		}
	}
}

# Country scope; FROM is the destination state.
FOE_relocate_nomadic_court = {
	if = {
		limit = { has_variable = FOE_nomadic_court_state }
		var:FOE_nomadic_court_state = {
			remove_dynamic_modifier = { modifier = FOE_nomadic_court_modifier scope = ROOT }
		}
	}
	set_capital = { state = FROM remember_old_capital = no }
	set_variable = { FOE_nomadic_court_state = FROM.id }
	FROM = {
		add_dynamic_modifier = { modifier = FOE_nomadic_court_modifier scope = ROOT }
	}
	set_country_flag = { flag = FOE_nomadic_court_cooldown days = 100  value = 1}
}

# Country scope; FROM remains the new court state after relocation.
FOE_integrate_nomadic_frontier = {
	FROM = {
		add_core_of = ROOT
		every_neighbor_state = {
			limit = {
				is_owned_by = ROOT
				NOT = { is_core_of = ROOT }
			}
			add_compliance = 10
			add_resistance = -20
		}
	}
}

#Slaving Despot Subclass
# Mythic: industry and occupied-state resistance both grow by 1% per thousand slaves.
FOE_update_empire_of_chains = {
	if = {
		limit = { FOE_is_mythic_slaving_despot = yes }
		set_variable = { FOE_chains_industry = foe_slaves_var }
		multiply_variable = { FOE_chains_industry = 0.01 }
		clamp_variable = { var = FOE_chains_industry min = 0 max = 0.25 }
		set_variable = { FOE_chains_resistance = foe_slaves_var }
		multiply_variable = { FOE_chains_resistance = 0.01 }
		clamp_variable = { var = FOE_chains_resistance min = 0 max = 0.50 }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = FOE_empire_of_chains_modifier } } }
			add_dynamic_modifier = { modifier = FOE_empire_of_chains_modifier }
		}
		force_update_dynamic_modifier = yes
	}
	else_if = {
		limit = { has_dynamic_modifier = { modifier = FOE_empire_of_chains_modifier } }
		remove_dynamic_modifier = { modifier = FOE_empire_of_chains_modifier }
		clear_variable = FOE_chains_industry
		clear_variable = FOE_chains_resistance
	}
}

# Country scope (new controller); FROM.FROM is the conquered state.
# One thousand slaves per 3,000 population, up to 5.
FOE_take_conquest_slaves = {
	FROM.FROM = {
		set_state_flag = { flag = FOE_despot_conquest_slaves_taken value = 1 days = 180 }
	}
	if = {
		limit = { FROM.FROM = { state_population > 14999 } }
		add_to_variable = { var = foe_slaves_var value = 5 }
	}
	else_if = {
		limit = { FROM.FROM = { state_population > 11999 } }
		add_to_variable = { var = foe_slaves_var value = 4 }
	}
	else_if = {
		limit = { FROM.FROM = { state_population > 8999 } }
		add_to_variable = { var = foe_slaves_var value = 3 }
	}
	else_if = {
		limit = { FROM.FROM = { state_population > 5999 } }
		add_to_variable = { var = foe_slaves_var value = 2 }
	}
	else_if = {
		limit = { FROM.FROM = { state_population > 2999 } }
		add_to_variable = { var = foe_slaves_var value = 1 }
	}
	FOE_update_empire_of_chains = yes
}

#Iron Warden Subclass
# State scope; ROOT is the Iron Warden. The work is done: the walls go up and the state is ours for good.
FOE_secure_state = {
	add_building_construction = {
		type = bunker
		level = 2
		instant_build = yes
		province = {
			all_provinces = yes
			limit_to_border = yes
		}
	}
	add_dynamic_modifier = { modifier = FOE_warden_bastion_modifier }
	clr_state_flag = FOE_warden_securing
	clr_state_flag = FOE_warden_auto_securing
	clear_variable = FOE_warden_work_progress
	# A Legendary Warden pacifies the population along with the ground.
	if = {
		limit = { owner = { FOE_is_legendary_or_mythic_iron_warden = yes } }
		add_compliance = 25
		cancel_resistance = yes
	}
}

# Country scope. The Iron Line counts secured states against everything we hold but have not locked down,
# so conquering faster than we consolidate is worth nothing.
FOE_update_iron_warden = {
	if = {
		limit = { FOE_is_iron_warden = yes }
		set_variable = { FOE_warden_secured_states = 0 }
		every_owned_state = {
			limit = { has_dynamic_modifier = { modifier = FOE_warden_bastion_modifier } }
			ROOT = { add_to_variable = { FOE_warden_secured_states = 1 } }
		}
		# Secured minus unsecured, which is twice the secured states less everything we own.
		set_temp_variable = { FOE_warden_line = FOE_warden_secured_states }
		multiply_temp_variable = { FOE_warden_line = 2 }
		subtract_from_temp_variable = { FOE_warden_line = num_owned_states }
		clamp_temp_variable = { var = FOE_warden_line min = 0 }
		if = {
			limit = { FOE_is_mythic_iron_warden = yes }
			multiply_temp_variable = { FOE_warden_line = 2 }
		}
		set_variable = { FOE_iron_line_defence = FOE_warden_line }
		multiply_variable = { FOE_iron_line_defence = 0.01 }
		clamp_variable = { var = FOE_iron_line_defence min = 0 max = 0.20 }
		set_variable = { FOE_iron_line_dig_in = FOE_warden_line }
		multiply_variable = { FOE_iron_line_dig_in = 0.25 }
		clamp_variable = { var = FOE_iron_line_dig_in min = 0 max = 5 }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = FOE_the_iron_line_modifier } } }
			add_dynamic_modifier = { modifier = FOE_the_iron_line_modifier }
		}
		if = {
			limit = {
				FOE_is_legendary_or_mythic_iron_warden = yes
				NOT = { has_idea = FOE_the_wardens_peace }
			}
			add_ideas = FOE_the_wardens_peace
		}
		if = {
			limit = {
				FOE_is_mythic_iron_warden = yes
				NOT = { has_idea = FOE_the_line_holds }
			}
			add_ideas = FOE_the_line_holds
		}
		# Unordered work still ties up builders, three factories per site, as the decision does.
		set_variable = { FOE_warden_works = 0 }
		every_owned_state = {
			limit = { has_state_flag = FOE_warden_auto_securing }
			ROOT = { add_to_variable = { FOE_warden_works = 1 } }
		}
		multiply_variable = { FOE_warden_works = 3 }
		if = {
			limit = {
				check_variable = { FOE_warden_works > 0 }
				NOT = { has_dynamic_modifier = { modifier = FOE_warden_works_modifier } }
			}
			add_dynamic_modifier = { modifier = FOE_warden_works_modifier }
		}
		else_if = {
			limit = {
				check_variable = { FOE_warden_works < 1 }
				has_dynamic_modifier = { modifier = FOE_warden_works_modifier }
			}
			remove_dynamic_modifier = { modifier = FOE_warden_works_modifier }
		}
	}
	else_if = {
		limit = { has_dynamic_modifier = { modifier = FOE_the_iron_line_modifier } }
		remove_dynamic_modifier = { modifier = FOE_the_iron_line_modifier }
		if = {
			limit = { has_dynamic_modifier = { modifier = FOE_warden_works_modifier } }
			remove_dynamic_modifier = { modifier = FOE_warden_works_modifier }
		}
		clear_variable = FOE_warden_secured_states
		clear_variable = FOE_warden_works
		clear_variable = FOE_warden_slots
		clear_variable = FOE_warden_sites
		clear_variable = FOE_iron_line_defence
		clear_variable = FOE_iron_line_dig_in
	}
}

# Country scope; a Mythic Iron Warden. Runs weekly. The engineers pick up every unsecured state by
# themselves and work it for seven weeks. The countdown lives on the state, because a state event is
# fired FOR a country and so arrives in country scope, carrying nothing about the state it is named for.
FOE_tick_warden_works = {
	if = {
		limit = { FOE_is_mythic_iron_warden = yes }
		# One building crew per 30 factories we own, never more than five sites at once.
		set_variable = { FOE_warden_slots = num_of_factories }
		set_variable = { FOE_warden_slot_remainder = FOE_warden_slots }
		modulo_variable = { var = FOE_warden_slot_remainder value = 30 }
		subtract_from_variable = { FOE_warden_slots = FOE_warden_slot_remainder }
		divide_variable = { FOE_warden_slots = 30 }
		clamp_variable = { var = FOE_warden_slots min = 0 max = 5 }
		clear_variable = FOE_warden_slot_remainder
		set_variable = { FOE_warden_sites = 0 }
		every_owned_state = {
			limit = { has_state_flag = FOE_warden_auto_securing }
			ROOT = { add_to_variable = { FOE_warden_sites = 1 } }
		}
		every_owned_state = {
			limit = {
				FOE_is_securable_state = yes
				ROOT = { check_variable = { FOE_warden_sites < FOE_warden_slots } }
			}
			set_state_flag = { flag = FOE_warden_auto_securing value = 1 days = 90 }
			set_variable = { FOE_warden_work_progress = 0 }
			ROOT = { add_to_variable = { FOE_warden_sites = 1 } }
		}
	}
	every_owned_state = {
		limit = { has_state_flag = FOE_warden_auto_securing }
		if = {
			limit = {
				owner = { FOE_is_mythic_iron_warden = yes }
				is_fully_controlled_by = OWNER
			}
			add_to_variable = { FOE_warden_work_progress = 1 }
			if = {
				limit = { check_variable = { FOE_warden_work_progress > 6 } }
				FOE_secure_state = yes
			}
		}
		else = {
			# The site is abandoned if we lose the ground, or the Warden who ordered the work.
			clr_state_flag = FOE_warden_auto_securing
			clear_variable = FOE_warden_work_progress
		}
	}
	FOE_update_iron_warden = yes
}

FOE_initialize_iron_warden = {
	add_country_leader_trait = FOE_iron_warden_trait
	var:country_leader = {
		set_character_flag = FOE_great_leader_character
		set_character_flag = FOE_iron_warden_character
	}
	FOE_update_iron_warden = yes
}

# Country scope. The stolen goods go back to work: a civilian factory where there is room for one.
FOE_chainbreaker_build_liberated_factory = {
	random_owned_controlled_state = {
		limit = {
			free_building_slots = {
				building = industrial_complex
				size > 0
				include_locked = yes
			}
			OR = {
				is_in_home_area = yes
				NOT = {
					owner = {
						any_owned_state = {
							free_building_slots = {
								building = industrial_complex
								size > 0
								include_locked = yes
							}
							is_in_home_area = yes
						}
					}
				}
			}
		}
		add_extra_state_shared_building_slots = 1
		add_building_construction = {
			type = industrial_complex
			level = 1
			instant_build = yes
		}
	}
}

# Country scope; ROOT is the Chainbreaker, FROM the previous controller, FROM.FROM the state.
# Frees as many thousands of slaves as the Despot would take from a state this size, but no more than FROM holds.
# Each thousand gives 200 manpower and 800 refugees. Also recovers up to 5 of a raider's loot; every 5 builds a factory.
FOE_chainbreaker_liberate_state = {
	FROM.FROM = { set_state_flag = { flag = FOE_chainbreaker_liberated value = 1 days = 180 } }
	if = {
		limit = { FROM = { check_variable = { foe_slaves_var > 0 } } }
		set_variable = { FOE_chainbreaker_freed = 0 }
		if = {
			limit = { FROM.FROM = { state_population > 14999 } }
			set_variable = { FOE_chainbreaker_freed = 5 }
		}
		else_if = {
			limit = { FROM.FROM = { state_population > 11999 } }
			set_variable = { FOE_chainbreaker_freed = 4 }
		}
		else_if = {
			limit = { FROM.FROM = { state_population > 8999 } }
			set_variable = { FOE_chainbreaker_freed = 3 }
		}
		else_if = {
			limit = { FROM.FROM = { state_population > 5999 } }
			set_variable = { FOE_chainbreaker_freed = 2 }
		}
		else_if = {
			limit = { FROM.FROM = { state_population > 2999 } }
			set_variable = { FOE_chainbreaker_freed = 1 }
		}
		set_variable = { FOE_chainbreaker_freed_cap = FROM.foe_slaves_var }
		if = {
			limit = { check_variable = { FOE_chainbreaker_freed > FOE_chainbreaker_freed_cap } }
			set_variable = { FOE_chainbreaker_freed = FOE_chainbreaker_freed_cap }
		}
		if = {
			limit = { check_variable = { FOE_chainbreaker_freed > 0 } }
			FROM = { subtract_from_variable = { foe_slaves_var = ROOT.FOE_chainbreaker_freed } }
			set_variable = { FOE_chainbreaker_payout = FOE_chainbreaker_freed }
			multiply_variable = { FOE_chainbreaker_payout = 200 }
			add_manpower = FOE_chainbreaker_payout
			multiply_variable = { FOE_chainbreaker_payout = 4 }
			if = {
				limit = { NOT = { has_variable = unassimilated_refugees } }
				set_variable = { unassimilated_refugees = 0 }
			}
			add_to_variable = { unassimilated_refugees = FOE_chainbreaker_payout }
			on_refugee_change = yes
		}
		clear_variable = FOE_chainbreaker_freed
		clear_variable = FOE_chainbreaker_freed_cap
		clear_variable = FOE_chainbreaker_payout
	}
	if = {
		limit = { FROM = { check_variable = { loot > 0 } } }
		set_variable = { FOE_chainbreaker_loot_taken = FROM.loot }
		clamp_variable = { var = FOE_chainbreaker_loot_taken min = 0 max = 5 }
		FROM = { subtract_from_variable = { loot = ROOT.FOE_chainbreaker_loot_taken } }
		if = {
			limit = { NOT = { has_variable = FOE_liberated_loot } }
			set_variable = { FOE_liberated_loot = 0 }
		}
		add_to_variable = { FOE_liberated_loot = FOE_chainbreaker_loot_taken }
		clear_variable = FOE_chainbreaker_loot_taken
		if = {
			limit = { check_variable = { FOE_liberated_loot > 4 } }
			subtract_from_variable = { FOE_liberated_loot = 5 }
			FOE_chainbreaker_build_liberated_factory = yes
		}
	}
}

#Architect of Order Subclass
# Country scope. Planned Capital follows the current capital and grows with its population.
# Mythic: The Ordered State grows with it too, and turns negative while the heart has fallen.
FOE_update_planned_capital = {
	if = {
		limit = { FOE_is_architect_of_order = yes }
		capital_scope = {
			ROOT = { set_temp_variable = { FOE_current_capital = PREV.id } }
		}
		if = {
			limit = {
				has_variable = FOE_planned_capital_state
				NOT = { check_variable = { FOE_planned_capital_state = FOE_current_capital } }
			}
			var:FOE_planned_capital_state = {
				remove_dynamic_modifier = { modifier = FOE_planned_capital_modifier }
			}
		}
		set_variable = { FOE_planned_capital_state = FOE_current_capital }
		if = {
			limit = { NOT = { has_country_flag = FOE_centre_fallen } }
			set_variable = { FOE_architect_heart_state = FOE_current_capital }
		}
		capital_scope = {
			# Per thousand residents: population growth +1%, resources +1%,
			# recruitable population +0.5% (max 10%), construction speed +1% (max 25%).
			set_variable = { FOE_planned_population = state_population_k }
			multiply_variable = { FOE_planned_population = 0.01 }
			set_variable = { FOE_planned_resources = state_population_k }
			multiply_variable = { FOE_planned_resources = 0.01 }
			set_variable = { FOE_planned_recruitable = state_population_k }
			multiply_variable = { FOE_planned_recruitable = 0.005 }
			clamp_variable = { var = FOE_planned_recruitable min = 0 max = 0.10 }
			set_variable = { FOE_planned_construction = state_population_k }
			multiply_variable = { FOE_planned_construction = 0.01 }
			clamp_variable = { var = FOE_planned_construction min = 0 max = 0.25 }
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = FOE_planned_capital_modifier } } }
				add_dynamic_modifier = { modifier = FOE_planned_capital_modifier }
			}
			force_update_dynamic_modifier = yes
		}
	}
	else_if = {
		limit = { has_variable = FOE_planned_capital_state }
		var:FOE_planned_capital_state = {
			remove_dynamic_modifier = { modifier = FOE_planned_capital_modifier }
		}
		clear_variable = FOE_planned_capital_state
		clear_variable = FOE_architect_heart_state
	}

	if = {
		limit = { FOE_is_mythic_architect_of_order = yes }
		capital_scope = {
			ROOT = { set_temp_variable = { FOE_capital_population_k = PREV.state_population_k } }
		}
		# Per thousand residents: stability +0.5% (max 15%), research speed +0.5% (max 10%).
		set_variable = { FOE_ordered_stability = FOE_capital_population_k }
		multiply_variable = { FOE_ordered_stability = 0.005 }
		clamp_variable = { var = FOE_ordered_stability min = 0 max = 0.15 }
		set_variable = { FOE_ordered_research = FOE_capital_population_k }
		multiply_variable = { FOE_ordered_research = 0.005 }
		clamp_variable = { var = FOE_ordered_research min = 0 max = 0.10 }
		if = {
			limit = { has_country_flag = FOE_centre_fallen }
			multiply_variable = { FOE_ordered_stability = -1 }
			multiply_variable = { FOE_ordered_research = -1 }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = FOE_ordered_state_modifier } } }
			add_dynamic_modifier = { modifier = FOE_ordered_state_modifier }
		}
		force_update_dynamic_modifier = yes
	}
	else_if = {
		limit = { has_dynamic_modifier = { modifier = FOE_ordered_state_modifier } }
		remove_dynamic_modifier = { modifier = FOE_ordered_state_modifier }
		clear_variable = FOE_ordered_stability
		clear_variable = FOE_ordered_research
	}
	if = {
		limit = {
			has_country_flag = FOE_centre_fallen
			NOT = { FOE_is_mythic_architect_of_order = yes }
		}
		clr_country_flag = FOE_centre_fallen
	}
}

	}

		set_state_category = city
	}
	else_if = {
		limit = { has_state_category = town }
		set_state_category = large_town
	}
	else_if = {
		limit = { has_state_category = shanty }
		set_state_category = town
	}
	else = {

	}

# Country scope; ROOT is the Perfectionist, FROM the state being ceded.
FOE_cede_periphery = {
	# First, the state's industry comes home. Dockyards need a coast, so they go offsite.
	FROM = {
		set_state_flag = FOE_being_ceded
		while_loop_effect = {
			limit = { check_variable = { industrial_complex_level > 0 } }
			remove_building = { type = industrial_complex level = 1 }
			ROOT = { FOE_rehouse_ceded_civilian_factory = yes }
		}
		while_loop_effect = {
			limit = { check_variable = { arms_factory_level > 0 } }
			remove_building = { type = arms_factory level = 1 }
			ROOT = { FOE_rehouse_ceded_military_factory = yes }
		}
		while_loop_effect = {
			limit = { check_variable = { building_level@dockyard > 0 } }
			remove_building = { type = dockyard level = 1 }
			ROOT = { add_offsite_building = { type = dockyard level = 1 } }
		}
	}
	# Second, the state goes to a neighbour we are at peace with, and we give up any claim to it.
	FROM = {
		if = {
			limit = { has_dynamic_modifier = { modifier = FOE_perfected_heartland_modifier } }
			remove_dynamic_modifier = { modifier = FOE_perfected_heartland_modifier }
		}
		clear_variable = FOE_heartland_level
		clear_variable = FOE_heartland_bonus
		clear_variable = FOE_heartland_core
		remove_claim_by = ROOT
		random_neighbor_state = {
			limit = { FOE_is_cede_recipient_state = yes }
			owner = { save_event_target_as = FOE_cede_recipient }
		}
		clr_state_flag = FOE_being_ceded
		transfer_state_to = event_target:FOE_cede_recipient
	}
	# Third, what we keep grows more perfect: a level 4 capital first, then any level 4 state,
	# then the capital, then any core state not yet at level 5.
	if = {
		limit = { capital_scope = { check_variable = { FOE_heartland_level = 4 } } }
		capital_scope = { FOE_raise_heartland = yes }
	}
	else_if = {
		limit = {
			any_owned_state = {
				is_core_of = ROOT
				check_variable = { FOE_heartland_level = 4 }
			}
		}
		random_owned_state = {
			limit = {
				is_core_of = ROOT
				check_variable = { FOE_heartland_level = 4 }
			}
			FOE_raise_heartland = yes
		}
	}
	else_if = {
		limit = { capital_scope = { NOT = { check_variable = { FOE_heartland_level > 4 } } } }
		capital_scope = { FOE_raise_heartland = yes }
	}
	else_if = {
		limit = {
			any_owned_state = {
				is_core_of = ROOT
				NOT = { check_variable = { FOE_heartland_level > 4 } }
			}
		}
		random_owned_state = {
			limit = {
				is_core_of = ROOT
				NOT = { check_variable = { FOE_heartland_level > 4 } }
			}
			FOE_raise_heartland = yes
		}
	}
}

# Country scope. Legendary: the resource with the largest shortfall (imports minus balance) gets a flat bonus of that size
# (at least 3, at most 12) for 180 days.
FOE_apply_import_substitution = {
	set_temp_variable = { FOE_largest_deficit = -1000 }
	set_temp_variable = { FOE_largest_deficit_resource = 0 }
	set_temp_variable = { FOE_resource_deficit = resource_imported@water }
	subtract_from_temp_variable = { FOE_resource_deficit = resource@water }
	if = {
		limit = {
			NOT = { FOE_is_self_sufficient_in_water = yes }
			check_variable = { FOE_resource_deficit > FOE_largest_deficit }
		}
		set_temp_variable = { FOE_largest_deficit = FOE_resource_deficit }
		set_temp_variable = { FOE_largest_deficit_resource = 1 }
	}
	set_temp_variable = { FOE_resource_deficit = resource_imported@energy }
	subtract_from_temp_variable = { FOE_resource_deficit = resource@energy }
	if = {
		limit = {
			NOT = { FOE_is_self_sufficient_in_energy = yes }
			check_variable = { FOE_resource_deficit > FOE_largest_deficit }
		}
		set_temp_variable = { FOE_largest_deficit = FOE_resource_deficit }
		set_temp_variable = { FOE_largest_deficit_resource = 2 }
	}
	set_temp_variable = { FOE_resource_deficit = resource_imported@metal }
	subtract_from_temp_variable = { FOE_resource_deficit = resource@metal }
	if = {
		limit = {
			NOT = { FOE_is_self_sufficient_in_metal = yes }
			check_variable = { FOE_resource_deficit > FOE_largest_deficit }
		}
		set_temp_variable = { FOE_largest_deficit = FOE_resource_deficit }
		set_temp_variable = { FOE_largest_deficit_resource = 3 }
	}
	set_temp_variable = { FOE_resource_deficit = resource_imported@composites }
	subtract_from_temp_variable = { FOE_resource_deficit = resource@composites }
	if = {
		limit = {
			NOT = { FOE_is_self_sufficient_in_composites = yes }
			check_variable = { FOE_resource_deficit > FOE_largest_deficit }
		}
		set_temp_variable = { FOE_largest_deficit = FOE_resource_deficit }
		set_temp_variable = { FOE_largest_deficit_resource = 4 }
	}
	set_temp_variable = { FOE_resource_deficit = resource_imported@circuitry }
	subtract_from_temp_variable = { FOE_resource_deficit = resource@circuitry }
	if = {
		limit = {
			NOT = { FOE_is_self_sufficient_in_circuitry = yes }
			check_variable = { FOE_resource_deficit > FOE_largest_deficit }
		}
		set_temp_variable = { FOE_largest_deficit = FOE_resource_deficit }
		set_temp_variable = { FOE_largest_deficit_resource = 5 }
	}
	clamp_temp_variable = { var = FOE_largest_deficit min = 3 max = 12 }
	if = {
		limit = { check_variable = { FOE_largest_deficit_resource = 1 } }
		set_variable = { FOE_import_substitution_water = FOE_largest_deficit }
		add_dynamic_modifier = { modifier = FOE_import_substitution_water_modifier days = 180 }
	}
	else_if = {
		limit = { check_variable = { FOE_largest_deficit_resource = 2 } }
		set_variable = { FOE_import_substitution_energy = FOE_largest_deficit }
		add_dynamic_modifier = { modifier = FOE_import_substitution_energy_modifier days = 180 }
	}
	else_if = {
		limit = { check_variable = { FOE_largest_deficit_resource = 3 } }
		set_variable = { FOE_import_substitution_metal = FOE_largest_deficit }
		add_dynamic_modifier = { modifier = FOE_import_substitution_metal_modifier days = 180 }
	}
	else_if = {
		limit = { check_variable = { FOE_largest_deficit_resource = 4 } }
		set_variable = { FOE_import_substitution_composites = FOE_largest_deficit }
		add_dynamic_modifier = { modifier = FOE_import_substitution_composites_modifier days = 180 }
	}
	else_if = {
		limit = { check_variable = { FOE_largest_deficit_resource = 5 } }
		set_variable = { FOE_import_substitution_circuitry = FOE_largest_deficit }
		add_dynamic_modifier = { modifier = FOE_import_substitution_circuitry_modifier days = 180 }
	}

}

# Country scope. Keeps the Autark in a closed economy (enclave laws may stay), and updates
# Autarky (Great) and Fortress Economy and Scavenger Networks (Mythic).
FOE_update_autark = {
	if = {
		limit = { FOE_is_autark = yes }
		if = {
			limit = {
				NOT = {
					has_idea = closed_economy
					has_idea = enclave_economy
					has_idea = lit_enclave_economy
				}
			}
			add_ideas = closed_economy
		}
		FOE_count_self_sufficient_resources = yes
		# Per self-sufficient resource: factory output +2%, stability +1%.
		set_variable = { FOE_autarky_output = FOE_autark_self_sufficient }
		multiply_variable = { FOE_autarky_output = 0.02 }
		set_variable = { FOE_autarky_stability = FOE_autark_self_sufficient }
		multiply_variable = { FOE_autarky_stability = 0.01 }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = FOE_autarky_modifier } } }
			add_dynamic_modifier = { modifier = FOE_autarky_modifier }
		}
		if = {
			limit = {
				FOE_is_mythic_autark = yes
				check_variable = { FOE_autark_self_sufficient > 4 }
			}
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = FOE_fortress_economy_modifier } } }
				add_dynamic_modifier = { modifier = FOE_fortress_economy_modifier }
			}
		}
		else_if = {
			limit = { has_dynamic_modifier = { modifier = FOE_fortress_economy_modifier } }
			remove_dynamic_modifier = { modifier = FOE_fortress_economy_modifier }
		}
		if = {
			limit = {
				FOE_is_mythic_autark = yes
				check_variable = { FOE_autark_self_sufficient < 5 }
				check_variable = { FOE_autark_scavenged < 20 }
			}
			random_list = {
				log = yes
				10 = { FOE_autark_scavenge = yes }
				90 = { }
			}
		}
		force_update_dynamic_modifier = yes
	}
	else_if = {
		limit = { has_dynamic_modifier = { modifier = FOE_autarky_modifier } }
		remove_dynamic_modifier = { modifier = FOE_autarky_modifier }
		if = {
			limit = { has_dynamic_modifier = { modifier = FOE_fortress_economy_modifier } }
			remove_dynamic_modifier = { modifier = FOE_fortress_economy_modifier }
		}
	}

			limit = { NOT = { has_idea = FOE_the_steel_dream } }
			add_ideas = FOE_the_steel_dream
		}
		if = {
			limit = { NOT = { has_variable = FOE_dreamer_exchanges } }
			set_variable = { FOE_dreamer_exchanges = 0 }
			FOE_update_dreamer_costs = yes
		}
		if = {
			limit = { FOE_is_machine_uprising = yes }
			FOE_create_uprising_templates = yes
			if = {
				limit = { NOT = { has_variable = FOE_uprising_swarm_count } }
				set_variable = { FOE_uprising_swarm_count = 0 }
			}
		}
		set_temp_variable = { FOE_weighted_robots = num_equipment@light_robot_equipment }
		set_temp_variable = { FOE_medium_robots = num_equipment@medium_robot_equipment }
		multiply_temp_variable = { FOE_medium_robots = 2 }
		add_to_temp_variable = { FOE_weighted_robots = FOE_medium_robots }
		set_temp_variable = { FOE_heavy_robots = num_equipment@heavy_robot_equipment }
		multiply_temp_variable = { FOE_heavy_robots = 4 }
		add_to_temp_variable = { FOE_weighted_robots = FOE_heavy_robots }

		set_variable = { FOE_automation_consumer_goods = FOE_weighted_robots }
		multiply_variable = { FOE_automation_consumer_goods = -0.0001 }
		clamp_variable = { var = FOE_automation_consumer_goods min = -0.10 max = 0 }
		set_variable = { FOE_automation_efficiency = 0 }
		if = {
			limit = { FOE_is_legendary_or_mythic_dreamer_in_steel = yes }
			set_variable = { FOE_automation_efficiency = FOE_weighted_robots }
			multiply_variable = { FOE_automation_efficiency = 0.0002 }
			clamp_variable = { var = FOE_automation_efficiency min = 0 max = 0.20 }
		}
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = FOE_automation_modifier } } }
			add_dynamic_modifier = { modifier = FOE_automation_modifier }
		}
		force_update_dynamic_modifier = yes

		# Once a civilian factory costs 200 robots, our industry is automated enough for the machines to rise.
		if = {
			limit = {
				NOT = { FOE_is_machine_uprising = yes }
				check_variable = { FOE_dreamer_civilian_cost > 199 }
			}
			random_list = {
				log = yes
				1 = { country_event = { id = special_rulers.11 } }
				99 = { }
			}
		}

			set_temp_variable = { FOE_replicated_robots = num_of_factories }
			divide_temp_variable = { FOE_replicated_robots = 2 }
			round_temp_variable = FOE_replicated_robots
			clamp_temp_variable = { var = FOE_replicated_robots min = 0 max = 50 }
			add_equipment_to_stockpile = { type = heavy_robot_industrial_equipment amount = FOE_replicated_robots }
		}
	}
	else_if = {
		limit = { has_dynamic_modifier = { modifier = FOE_automation_modifier } }
		remove_dynamic_modifier = { modifier = FOE_automation_modifier }
	}
}

# State scope; ROOT is the Machine Uprising. Every tenth swarm also builds an Uprising Colossus.

	}
}

# Country scope; the Machine Uprising. Ten days in: war on every neighbour, and a first wave of machines.
FOE_machine_uprising_mobilisation = {
	FOE_create_uprising_templates = yes
	every_neighbor_country = {
		limit = {
			exists = yes
			NOT = { has_war_with = ROOT }
		}
		ROOT = {
			declare_war_on = {
				target = PREV
				type = annex_everything
			}
		}
	}
	set_temp_variable = { FOE_uprising_wave = 20 }
	while_loop_effect = {
		limit = { check_variable = { FOE_uprising_wave > 0 } }
		random_owned_controlled_state = {
			create_unit = {
				division = "name = \"Uprising Swarm\" division_template = \"Uprising Swarm\" start_experience_factor = 0.3 start_equipment_factor = 0.9"
				owner = ROOT
			}
		}
		subtract_from_temp_variable = { FOE_uprising_wave = 1 }
	}
	set_temp_variable = { FOE_uprising_wave = 10 }
	while_loop_effect = {
		limit = { check_variable = { FOE_uprising_wave > 0 } }
		random_owned_controlled_state = {
			create_unit = {
				division = "name = \"Uprising Colossus\" division_template = \"Uprising Colossus\" start_experience_factor = 0.5 start_equipment_factor = 1.0"
				owner = ROOT
			}
		}
		subtract_from_temp_variable = { FOE_uprising_wave = 1 }
	}
}

# Any scope. Once the Uprising is gone from the map, nothing should still be claimed in its name.
FOE_clean_up_machine_uprising = {
	if = {
		limit = {
			has_global_flag = FOE_machine_uprising_happened
			NOT = { has_global_flag = FOE_machine_uprising_cleaned_up }
			NOT = { XYZ = { exists = yes } }
		}
		set_global_flag = FOE_machine_uprising_cleaned_up
		every_state = {
			limit = {
				OR = {
					is_core_of = XYZ
					is_claimed_by = XYZ
				}
			}
			remove_core_of = XYZ
			remove_claim_by = XYZ
		}
	}
}

FOE_machine_uprising_spread = {
	if = {
		limit = {
			any_owned_state = {
				is_controlled_by = ROOT
				NOT = { is_capital = yes }
			}
		}
		random_owned_controlled_state = {
			limit = { NOT = { is_capital = yes } }
			transfer_state_to = XYZ
		}
	}
	transfer_units_fraction = {
		target = XYZ
		size = 0
		army_ratio = 0.3
	}
}

FOE_start_machine_uprising = {
	set_global_flag = FOE_machine_uprising_happened
	clr_global_flag = FOE_machine_uprising_cleaned_up
	if = {
		limit = {
			any_owned_state = {
				is_controlled_by = ROOT
				NOT = { is_capital = yes }
			}
		}
		random_owned_controlled_state = {
			limit = { NOT = { is_capital = yes } }
			set_variable = { global.FOE_uprising_state = THIS.id }
			transfer_state_to = XYZ
			add_core_of = XYZ
		}
	}
	else = {
		capital_scope = {
			set_variable = { global.FOE_uprising_state = THIS.id }
			transfer_state_to = XYZ
			add_core_of = XYZ
		}
	}
	XYZ = {
		inherit_technology = ROOT
		set_capital = { state = global.FOE_uprising_state remember_old_capital = no }
		# Lets the Uprising use the Great Leader decision category.
		var:country_leader = { set_character_flag = FOE_great_leader_character }
		FOE_create_uprising_templates = yes
		if = {
			limit = { NOT = { has_variable = FOE_uprising_swarm_count } }
			set_variable = { FOE_uprising_swarm_count = 0 }
		}
	}
	transfer_units_fraction = {
		target = XYZ
		size = 0
		army_ratio = 0.5
	}
	if = {
		limit = { NOT = { XYZ = { has_war_with = ROOT } } }
		XYZ = {
			declare_war_on = {
				target = ROOT
				type = annex_everything
			}
		}
	}
	clear_variable = global.FOE_uprising_state
	every_country = {
		limit = {
			FOE_is_dreamer_in_steel = yes
			NOT = { FOE_is_machine_uprising = yes }
		}
		country_event = { id = special_rulers.13 days = 1 }
	}
	XYZ = { country_event = { id = special_rulers.14 days = 10 } }
}
