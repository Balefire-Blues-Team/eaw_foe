# Country scope. No scripted-effect parameters are used.

#Nomadic Khan Subclass
FOE_initialize_nomadic_khan = {
	add_country_leader_trait = FOE_nomadic_khan_trait
	var:country_leader = {
		set_character_flag = FOE_great_leader_character
		set_character_flag = FOE_nomadic_khan_character
	}
	if = {
		limit = { NOT = { has_variable = FOE_nomadic_court_state } }
		capital_scope = {
			ROOT = { set_variable = { FOE_nomadic_court_state = PREV.id } }
			add_dynamic_modifier = { modifier = FOE_nomadic_court_modifier scope = ROOT }
		}
	}
}

# Country scope; FROM is the destination state.
FOE_relocate_nomadic_court = {
	if = {
		limit = { has_variable = FOE_nomadic_court_state }
		var:FOE_nomadic_court_state = {
			remove_dynamic_modifier = { modifier = FOE_nomadic_court_modifier scope = ROOT }
		}
	}
	set_capital = { state = FROM remember_old_capital = no }
	set_variable = { FOE_nomadic_court_state = FROM.id }
	FROM = {
		add_dynamic_modifier = { modifier = FOE_nomadic_court_modifier scope = ROOT }
	}
	set_country_flag = { flag = FOE_nomadic_court_cooldown days = 100  value = 1}
}

# Country scope; FROM remains the new court state after relocation.
FOE_integrate_nomadic_frontier = {
	FROM = {
		add_core_of = ROOT
		every_neighbor_state = {
			limit = {
				is_owned_by = ROOT
				NOT = { is_core_of = ROOT }
			}
			add_compliance = 10
			add_resistance = -20
		}
	}
}

#Slaving Despot Subclass
# Mythic: industry and occupied-state resistance both grow by 1% per thousand slaves.
FOE_update_empire_of_chains = {
	if = {
		limit = { FOE_is_mythic_slaving_despot = yes }
		set_variable = { FOE_chains_industry = foe_slaves_var }
		multiply_variable = { FOE_chains_industry = 0.01 }
		clamp_variable = { var = FOE_chains_industry min = 0 max = 0.25 }
		set_variable = { FOE_chains_resistance = foe_slaves_var }
		multiply_variable = { FOE_chains_resistance = 0.01 }
		clamp_variable = { var = FOE_chains_resistance min = 0 max = 0.50 }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = FOE_empire_of_chains_modifier } } }
			add_dynamic_modifier = { modifier = FOE_empire_of_chains_modifier }
		}
		force_update_dynamic_modifier = yes
	}
	else_if = {
		limit = { has_dynamic_modifier = { modifier = FOE_empire_of_chains_modifier } }
		remove_dynamic_modifier = { modifier = FOE_empire_of_chains_modifier }
		clear_variable = FOE_chains_industry
		clear_variable = FOE_chains_resistance
	}
}

# Country scope (new controller); FROM.FROM is the conquered state.
# One thousand slaves per 3,000 population, up to 5.
FOE_take_conquest_slaves = {
	FROM.FROM = {
		set_state_flag = { flag = FOE_despot_conquest_slaves_taken value = 1 days = 180 }
	}
	if = {
		limit = { FROM.FROM = { state_population > 14999 } }
		add_to_variable = { var = foe_slaves_var value = 5 }
	}
	else_if = {
		limit = { FROM.FROM = { state_population > 11999 } }
		add_to_variable = { var = foe_slaves_var value = 4 }
	}
	else_if = {
		limit = { FROM.FROM = { state_population > 8999 } }
		add_to_variable = { var = foe_slaves_var value = 3 }
	}
	else_if = {
		limit = { FROM.FROM = { state_population > 5999 } }
		add_to_variable = { var = foe_slaves_var value = 2 }
	}
	else_if = {
		limit = { FROM.FROM = { state_population > 2999 } }
		add_to_variable = { var = foe_slaves_var value = 1 }
	}
	FOE_update_empire_of_chains = yes
}

# Nations without slavery gain access to it through their Despot.
FOE_initialize_slaving_despot = {
	add_country_leader_trait = FOE_slaving_despot_trait
	var:country_leader = {
		set_character_flag = FOE_great_leader_character
		set_character_flag = FOE_slaving_despot_character
	}
	set_country_flag = slavery_allowed
	if = {
		limit = { NOT = { has_variable = foe_slaves_var } }
		set_variable = { foe_slaves_var = 0 }
	}
	FOE_update_empire_of_chains = yes
}

#Carrion Marshal Subclass
FOE_create_carrion_templates = {
	if = {
		limit = { NOT = { has_country_flag = FOE_carrion_templates_created } }
		set_country_flag = FOE_carrion_templates_created
		division_template = {
			name = "Carrion Horde"
			is_locked = yes
			regiments = {
				militia = { x = 0 y = 0 }
				militia = { x = 0 y = 1 }
				militia = { x = 0 y = 2 }
				militia = { x = 0 y = 3 }
				militia = { x = 1 y = 0 }
				militia = { x = 1 y = 1 }
				militia = { x = 1 y = 2 }
				militia = { x = 1 y = 3 }
			}
		}
		division_template = {
			name = "Carrion Vanguard"
			is_locked = yes
			regiments = {
				spec_ops = { x = 0 y = 0 }
				spec_ops = { x = 0 y = 1 }
				spec_ops = { x = 0 y = 2 }
				spec_ops = { x = 0 y = 3 }
				spec_ops = { x = 0 y = 4 }
				infantry = { x = 1 y = 0 }
				infantry = { x = 1 y = 1 }
				infantry = { x = 1 y = 2 }
				infantry = { x = 1 y = 3 }
				infantry = { x = 1 y = 4 }
			}
			support = {
				fireteam_support = {
					x = 0
					y = 0
				}
				demolitions_support = {
					x = 0
					y = 1
				}
			}
		}
	}
}

# State scope; ROOT is the Marshal's country. Every tenth horde also raises a Carrion Vanguard.
FOE_spawn_carrion_horde = {
	create_unit = {
		division = "name = \"Carrion Horde\" division_template = \"Carrion Horde\" start_experience_factor = 0.3 start_equipment_factor = 0.9"
		owner = ROOT
	}
	ROOT = { add_to_variable = { FOE_carrion_horde_count = 1 } }
	if = {
		limit = { ROOT = { check_variable = { FOE_carrion_horde_count > 9 } } }
		create_unit = {
			division = "name = \"Carrion Vanguard\" division_template = \"Carrion Vanguard\" start_experience_factor = 0.5 start_equipment_factor = 1.0"
			owner = ROOT
		}
		ROOT = { set_variable = { FOE_carrion_horde_count = 0 } }
	}
}

# State scope; ROOT is the harvesting country.
# Consumes 2,000 population and one civilian factory if present; military industry is left intact.
FOE_carrion_harvest_state = {
	add_manpower = -2000
	add_dynamic_modifier = { modifier = FOE_carrion_devastation days = 365 }
	if = {
		limit = { check_variable = { industrial_complex_level > 0 } }
		remove_building = { type = industrial_complex level = 1 }
		FOE_spawn_carrion_horde = yes
	}
	FOE_spawn_carrion_horde = yes
}

#Unifier Subclass
# State scope; ROOT is the Unifier. Compliance is 5 plus 1 per core state, up to 30.
FOE_unifier_integrate_state = {
	set_temp_variable = { FOE_unifier_compliance = ROOT.num_core_states }
	add_to_temp_variable = { FOE_unifier_compliance = 5 }
	clamp_temp_variable = { var = FOE_unifier_compliance min = 5 max = 30 }
	add_compliance = FOE_unifier_compliance
	if = {
		limit = { compliance > 49 }
		add_core_of = ROOT
	}
}

# Legendary: stability and construction +1%, recruitable population +0.5% per core state.
FOE_update_one_nation = {
	if = {
		limit = { FOE_is_legendary_or_mythic_unifier = yes }
		set_variable = { FOE_unity_stability = num_core_states }
		multiply_variable = { FOE_unity_stability = 0.01 }
		clamp_variable = { var = FOE_unity_stability min = 0 max = 0.15 }
		set_variable = { FOE_unity_construction = num_core_states }
		multiply_variable = { FOE_unity_construction = 0.01 }
		clamp_variable = { var = FOE_unity_construction min = 0 max = 0.15 }
		set_variable = { FOE_unity_recruitable = num_core_states }
		multiply_variable = { FOE_unity_recruitable = 0.005 }
		clamp_variable = { var = FOE_unity_recruitable min = 0 max = 0.05 }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = FOE_one_nation_modifier } } }
			add_dynamic_modifier = { modifier = FOE_one_nation_modifier }
		}
		force_update_dynamic_modifier = yes
	}
	else_if = {
		limit = { has_dynamic_modifier = { modifier = FOE_one_nation_modifier } }
		remove_dynamic_modifier = { modifier = FOE_one_nation_modifier }
		clear_variable = FOE_unity_stability
		clear_variable = FOE_unity_construction
		clear_variable = FOE_unity_recruitable
	}
}

# Country scope; FROM is the absorbed country. Its states arrive as cores.
FOE_unifier_absorb_country = {
	FROM = {
		every_owned_state = { add_core_of = ROOT }
	}
	annex_country = { target = FROM transfer_troops = yes }
	set_country_flag = { flag = FOE_unifier_absorb_cooldown value = 1 days = 1095 }
	FOE_update_one_nation = yes
}
