# Keep current Politics is an effect, you'll need to put `keep_current_politics = yes` just before the puppet command, change for your ideologies as needed
keep_current_politics = {
	hidden_effect = {
		set_variable = { save_visionary = party_popularity@visionary }
		set_variable = { save_survivalism = party_popularity@survivalism }
		set_variable = { save_destruction = party_popularity@destruction }
		set_variable = { save_continuation = party_popularity@continuation }
		
		if = {
			limit = { has_government = visionary }
			set_country_flag = {
				flag = was_visionary
				value = 1
				days = 1
			}
		}
		if = {
			limit = { has_government = survivalism }
			set_country_flag = {
				flag = was_survivalism
				value = 1
				days = 1
			}
		}
		if = {
			limit = { has_government = destruction }
			set_country_flag = {
				flag = was_destruction
				value = 1
				days = 1
			}
		}
		if = {
			limit = { has_government = continuation }
			set_country_flag = {
				flag = was_continuation
				value = 1
				days = 1
			}
		}
		
		country_event = { id = EAI_M.8 hours = 1 }
	}
}

# Armoury Doctrine weapon kits (EaW's method): a special unit's rifles come from a hidden kit tech, granted
# with the unit and removed once its MEW conversion is researched. A negative need would keep the rifles'
# stats. Runs from the MEW decisions, the unlock techs and a weekly fallback (foe_on_actions.txt).
FOE_fix_sf_weapon_kits = {
	# power_armour
	if = {
		limit = {
			has_tech = stripped_pa_tech
			NOT = { has_tech = pa_mew_conversion_tech }
			NOT = { has_tech = pa_rifle_kit_tech }
		}
		set_technology = { pa_rifle_kit_tech = 1 popup = no }
	}
	if = {
		limit = {
			has_tech = pa_mew_conversion_tech
			has_tech = pa_rifle_kit_tech
		}
		set_technology = { pa_rifle_kit_tech = 0 popup = no }
	}
	# spec_ops
	if = {
		limit = {
			has_tech = spec_ops_equipment_tech_1
			NOT = { has_tech = spec_ops_mew_conversion_tech }
			NOT = { has_tech = spec_ops_rifle_kit_tech }
		}
		set_technology = { spec_ops_rifle_kit_tech = 1 popup = no }
	}
	if = {
		limit = {
			has_tech = spec_ops_mew_conversion_tech
			has_tech = spec_ops_rifle_kit_tech
		}
		set_technology = { spec_ops_rifle_kit_tech = 0 popup = no }
	}
	# pegasi
	if = {
		limit = {
			has_tech = tech_pegasi
			NOT = { has_tech = pegasi_mew_conversion_tech }
			NOT = { has_tech = pegasi_rifle_kit_tech }
		}
		set_technology = { pegasi_rifle_kit_tech = 1 popup = no }
	}
	if = {
		limit = {
			has_tech = pegasi_mew_conversion_tech
			has_tech = pegasi_rifle_kit_tech
		}
		set_technology = { pegasi_rifle_kit_tech = 0 popup = no }
	}
	# legionnaires
	if = {
		limit = {
			has_tech = tech_legion
			NOT = { has_tech = legion_mew_conversion_tech }
			NOT = { has_tech = legion_rifle_kit_tech }
		}
		set_technology = { legion_rifle_kit_tech = 1 popup = no }
	}
	if = {
		limit = {
			has_tech = legion_mew_conversion_tech
			has_tech = legion_rifle_kit_tech
		}
		set_technology = { legion_rifle_kit_tech = 0 popup = no }
	}
	# unicorn_company
	if = {
		limit = {
			has_tech = tech_unicorn
			NOT = { has_tech = unicorn_mew_conversion_tech }
			NOT = { has_tech = unicorn_rifle_kit_tech }
		}
		set_technology = { unicorn_rifle_kit_tech = 1 popup = no }
	}
	if = {
		limit = {
			has_tech = unicorn_mew_conversion_tech
			has_tech = unicorn_rifle_kit_tech
		}
		set_technology = { unicorn_rifle_kit_tech = 0 popup = no }
	}
}
