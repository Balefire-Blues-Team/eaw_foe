# Country scope: the template for GR_spawn_raider_mob, created once
GR_prepare_raider_mob = {
	if = {
		limit = {
			NOT = { has_country_flag = GR_raider_mob_template_flag }
		}
		set_country_flag = GR_raider_mob_template_flag
		division_template = {
			name = "Raider Mob"
			priority = 0
			is_locked = yes
			regiments = {
				infantry = { x = 0 y = 0 }
				infantry = { x = 0 y = 1 }
				infantry = { x = 0 y = 2 }
				infantry = { x = 0 y = 3 }
			}
		}
	}
}

# State scope, ROOT is the gang
GR_spawn_raider_mob = {
	create_unit = {
		division = "name = \"Raider Mob\" division_template = \"Raider Mob\" start_experience_factor = 0.2"
		owner = ROOT
	}
}
