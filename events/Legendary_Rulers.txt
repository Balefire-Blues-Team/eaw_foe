#Initiate Ruler choice
country_event = {
	id = special_rulers.1
	title = special_rulers.1.t
	desc = special_rulers.1.d
	picture = GFX_event_NCR_startup
	is_triggered_only = yes
	trigger = { FOE_great_leader_roll_eligible = yes }
	immediate = { FOE_mark_great_leader_roll_completed = yes }
	fire_only_once = no

	option = {
		name = special_rulers.1.a
		# Multipliers instead of negative adds, so the weight can never drop to zero or below.
		ai_chance = {
			base = 20
			modifier = {
				factor = 0.5
				is_major = yes
			}
			modifier = {
				factor = 0.85
				num_of_civilian_factories > 7
			}
			modifier = {
				factor = 0.85
				num_of_military_factories > 7
			}
			modifier = {
				factor = 0.5
				any_neighbor_country = {
					is_ai = no
				}
			}
		}
		country_event = { id = special_rulers.2}
		set_country_flag = hyper_aggressive_ai
		set_country_flag = great_leader_flag
	}
	option = {
		name = special_rulers.1.b
		ai_chance = { base = 80 }
	}
}

#Debug: fire with "event special_rulers.16" to make our own leader a Mythic Iron Warden.
country_event = {
	id = special_rulers.16
	title = special_rulers.16.t
	desc = special_rulers.16.d
	picture = GFX_event_NCR_startup
	is_triggered_only = yes
	fire_only_once = no

	option = {
		name = special_rulers.16.a
		trigger = {
			NOT = { has_country_leader_with_trait = mythic_leader_exploiter_trait }
		}
		# Swaps any class trait, Mythic included, for the Exploiter's.
		FOE_remove_class_traits = yes
		if = { limit = { has_country_leader_with_trait = mythic_leader_military_trait } remove_country_leader_trait = mythic_leader_military_trait }
		if = { limit = { has_country_leader_with_trait = mythic_leader_admin_trait } remove_country_leader_trait = mythic_leader_admin_trait }
		if = { limit = { has_country_leader_with_trait = mythic_leader_industry_trait } remove_country_leader_trait = mythic_leader_industry_trait }
		add_country_leader_trait = mythic_leader_exploiter_trait
		FOE_initialize_whisper_broker = yes
	}

	option = {
		name = special_rulers.16.b
		trigger = {
			has_country_leader_with_trait = mythic_leader_exploiter_trait
			NOT = { FOE_is_whisper_broker = yes }
		}
		FOE_initialize_whisper_broker = yes
	}

	# A full Dossier, a Black File on a random neighbour if we have none, both infiltration tokens against it,
	# and the Alert lifted. Network strength and operatives still have to be built normally.
	option = {
		name = special_rulers.16.c
		trigger = {
			FOE_is_whisper_broker = yes
		}
		if = {
			limit = {
				NOT = { FOE_has_valid_black_file = yes }
				any_neighbor_country = { FOE_is_black_file_candidate = yes }
			}
			random_neighbor_country = {
				limit = { FOE_is_black_file_candidate = yes }
				ROOT = { set_variable = { FOE_black_file = PREV } }
			}
		}
		set_variable = { FOE_dossier = 6 }
		FOE_refresh_insider_knowledge = yes
		clr_country_flag = FOE_broker_alerted
		if = {
			limit = { FOE_has_valid_black_file = yes }
			var:FOE_black_file = {
				ROOT = {
					add_operation_token = { tag = PREV token = token_civilian }
					add_operation_token = { tag = PREV token = token_army }
				}
			}
		}
	}

	option = {
		name = special_rulers.16.e
		trigger = {
			NOT = { has_country_leader_with_trait = mythic_leader_magnate_trait }
		}
		# Swaps any class trait, Mythic included, for the Magnate's.
		FOE_remove_class_traits = yes
		if = { limit = { has_country_leader_with_trait = mythic_leader_military_trait } remove_country_leader_trait = mythic_leader_military_trait }
		if = { limit = { has_country_leader_with_trait = mythic_leader_admin_trait } remove_country_leader_trait = mythic_leader_admin_trait }
		if = { limit = { has_country_leader_with_trait = mythic_leader_industry_trait } remove_country_leader_trait = mythic_leader_industry_trait }
		add_country_leader_trait = mythic_leader_magnate_trait
		FOE_initialize_deep_mine_sovereign = yes
	}

	option = {
		name = special_rulers.16.f
		trigger = {
			has_country_leader_with_trait = mythic_leader_magnate_trait
			NOT = { FOE_is_deep_mine_sovereign = yes }
		}
		FOE_initialize_deep_mine_sovereign = yes
	}

	# Finishes one dig now, free: depth +1, a slot, and the strike roll, with the report event.
	option = {
		name = special_rulers.16.g
		trigger = {
			FOE_is_deep_mine_sovereign = yes
			check_variable = { FOE_mine_depth_shown < 10 }
		}
		FOE_sink_the_shaft = yes
	}

	# Jumps the mine to Depth 10 so the Bottomless Mine starts rolling. Skipped digs grant no slots or finds.
	option = {
		name = special_rulers.16.k
		trigger = {
			FOE_is_deep_mine_sovereign = yes
			check_variable = { FOE_mine_depth_shown < 10 }
		}
		var:FOE_mine_state = { set_variable = { FOE_mine_depth = 10 } }
		FOE_update_deep_mine_sovereign = yes
	}

	option = {
		name = special_rulers.16.h
		trigger = {
			NOT = { has_country_leader_with_trait = mythic_leader_ruler_trait }
		}
		# Swaps any class trait, Mythic included, for the Ruler's.
		FOE_remove_class_traits = yes
		if = { limit = { has_country_leader_with_trait = mythic_leader_military_trait } remove_country_leader_trait = mythic_leader_military_trait }
		if = { limit = { has_country_leader_with_trait = mythic_leader_admin_trait } remove_country_leader_trait = mythic_leader_admin_trait }
		if = { limit = { has_country_leader_with_trait = mythic_leader_industry_trait } remove_country_leader_trait = mythic_leader_industry_trait }
		add_country_leader_trait = mythic_leader_ruler_trait
		FOE_initialize_just_monarch = yes
	}

	# Without a General candidate the Monarch aborts at once and the Ruler subclass is rerolled.
	option = {
		name = special_rulers.16.i
		trigger = {
			has_country_leader_with_trait = mythic_leader_ruler_trait
			NOT = { FOE_is_just_monarch = yes }
		}
		FOE_initialize_just_monarch = yes
	}

	# Bestow Honours becomes available now.
	option = {
		name = special_rulers.16.j
		trigger = {
			FOE_is_just_monarch = yes
			has_country_flag = FOE_honours_cooldown
		}
		clr_country_flag = FOE_honours_cooldown
	}

	# Breaks the bond as a leader change would, without the 180-day wait: the next weekly update picks a new General.
	option = {
		name = special_rulers.16.l
		trigger = {
			FOE_is_just_monarch = yes
			has_variable = FOE_loyal_general
		}
		FOE_break_sworn_bond = yes
		clr_country_flag = FOE_monarch_repick_delay

#DEBUG: Nemesis menu (console "event special_rulers.28"). Debug paths skip the roll gate.
country_event = {
	id = special_rulers.28
	title = special_rulers.28.t
	desc = special_rulers.28.d
	picture = GFX_event_NCR_startup
	is_triggered_only = yes
	fire_only_once = no

	# Swaps any class trait, Mythic included, for Great Conqueror, so the victory rank-up can be seen.
	# Needs a bordering AI Rival candidate; without one the log says "found no Rival" and nothing else happens.
	option = {
		name = special_rulers.28.a
		trigger = {
			NOT = { FOE_has_active_reckoning = yes }
			NOT = { FOE_is_nemesis = yes }
		}
		FOE_remove_class_traits = yes
		if = { limit = { has_country_leader_with_trait = mythic_leader_military_trait } remove_country_leader_trait = mythic_leader_military_trait }
		if = { limit = { has_country_leader_with_trait = mythic_leader_admin_trait } remove_country_leader_trait = mythic_leader_admin_trait }
		if = { limit = { has_country_leader_with_trait = mythic_leader_industry_trait } remove_country_leader_trait = mythic_leader_industry_trait }
		add_country_leader_trait = great_leader_military_trait
		FOE_initialize_nemesis = yes
	}

	option = {
		name = special_rulers.28.b
		trigger = { FOE_is_preparing_for_reckoning = yes }
		set_variable = { FOE_nemesis_preparation = 20 }
	}

	# Day 800 now: the pact ends, war is declared (or a draw is called) and Preparation is spent.
	option = {
		name = special_rulers.28.c
		trigger = { FOE_is_preparing_for_reckoning = yes }
		FOE_nemesis_reckoning_due = yes
	}

	# As if the Rival had capitulated, then a white peace ends the war. The weekly update then resolves both sides:
	# we win, and the Rival survives as the loser.
	option = {
		name = special_rulers.28.e
		trigger = {
			FOE_has_active_reckoning = yes
			has_country_flag = FOE_nemesis_war_started
			has_war_with = var:FOE_nemesis_rival
		}
		set_country_flag = FOE_nemesis_rival_fell
		var:FOE_nemesis_rival = {
			set_country_flag = FOE_nemesis_we_fell
			PREV = { white_peace = PREV }
		}
	}

	# Every Rival state becomes ours, so the Rival is gone. Next week we win; a Mythic victor cores them.
	option = {
		name = special_rulers.28.f
		trigger = {
			FOE_has_active_reckoning = yes
			has_country_flag = FOE_nemesis_war_started
			var:FOE_nemesis_rival = { exists = yes }
		}
		var:FOE_nemesis_rival = {
			every_owned_state = { transfer_state_to = ROOT }
		}
	}

	# One Conqueror rank up (Great, Legendary, Mythic), e.g. to test a Mythic victor's cores.
	option = {
		name = special_rulers.28.g
		trigger = {
			FOE_is_nemesis = yes
			NOT = { has_country_leader_with_trait = mythic_leader_military_trait }
		}
		FOE_nemesis_rank_up = yes
		FOE_update_nemesis_rank = yes
	}

	# Runs this week's update now instead of waiting.
	option = {
		name = special_rulers.28.h
		trigger = { FOE_has_active_reckoning = yes }
		FOE_update_nemesis = yes
	}

	option = { name = special_rulers.28.close }
}
