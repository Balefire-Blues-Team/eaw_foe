#Initiate Ruler choice
country_event = {
	id = special_rulers.1
	title = special_rulers.1.t
	desc = special_rulers.1.d
	picture = GFX_event_NCR_startup
	is_triggered_only = yes
	trigger = { FOE_great_leader_roll_eligible = yes }
	immediate = { set_country_flag = FOE_great_leader_roll_completed }
	fire_only_once = no

	option = {
		name = special_rulers.1.a
		# Multipliers instead of negative adds, so the weight can never drop to zero or below.
		ai_chance = {
			base = 20
			modifier = {
				factor = 0.5
				is_major = yes
			}
			modifier = {
				factor = 0.85
				num_of_civilian_factories > 7
			}
			modifier = {
				factor = 0.85
				num_of_military_factories > 7
			}
			modifier = {
				factor = 0.5
				any_neighbor_country = {
					is_ai = no
				}
			}
		}
		country_event = { id = special_rulers.2}
		set_country_flag = hyper_aggressive_ai
		set_country_flag = great_leader_flag
	}
	option = {
		name = special_rulers.1.b
		ai_chance = { base = 80 }
	}
}

#Debug: fire with "event special_rulers.16" to make our own leader a Mythic Iron Warden.
country_event = {
	id = special_rulers.16
	title = special_rulers.16.t
	desc = special_rulers.16.d
	picture = GFX_event_NCR_startup
	is_triggered_only = yes
	fire_only_once = no

	option = {
		name = special_rulers.16.a
		trigger = {
			NOT = { has_country_leader_with_trait = mythic_leader_military_trait }
		}
		add_country_leader_trait = mythic_leader_military_trait
		FOE_initialize_iron_warden = yes
	}

	option = {
		name = special_rulers.16.b
		trigger = {
			has_country_leader_with_trait = mythic_leader_military_trait
			NOT = { FOE_is_iron_warden = yes }
		}
		FOE_initialize_iron_warden = yes
	}

	# Saves waiting for the weekly tick when testing the works. Seven of these finish a site.
	option = {
		name = special_rulers.16.c
		trigger = {
			FOE_is_iron_warden = yes
		}
		FOE_tick_warden_works = yes
	}

	option = {
		name = special_rulers.16.e
		trigger = {
			NOT = { has_country_leader_with_trait = mythic_leader_admin_trait }
		}
		add_country_leader_trait = mythic_leader_admin_trait
		FOE_initialize_wild_heart = yes
	}

	option = {
		name = special_rulers.16.f
		trigger = {
			has_country_leader_with_trait = mythic_leader_admin_trait
			NOT = { FOE_is_wild_heart = yes }
		}
		FOE_initialize_wild_heart = yes
	}

	# Runs one migration now, exactly as the monthly tick does. Six of these reach a Legendary factory.
	option = {
		name = special_rulers.16.g
		trigger = {
			FOE_is_wild_heart = yes
		}
		FOE_wild_heart_migration = yes
	}

	option = {
		name = special_rulers.16.h
		trigger = {
			NOT = { has_country_leader_with_trait = mythic_leader_military_trait }
		}
		add_country_leader_trait = mythic_leader_military_trait
		FOE_initialize_chainbreaker = yes
	}

	option = {
		name = special_rulers.16.i
		trigger = {
			has_country_leader_with_trait = mythic_leader_military_trait
			NOT = { FOE_is_chainbreaker = yes }
		}
		FOE_initialize_chainbreaker = yes
	}

	# Enough refugees for one Freed March, through the exodus system's own refresh.
	option = {
		name = special_rulers.16.j
		trigger = {
			FOE_is_chainbreaker = yes
		}
		if = {
			limit = { NOT = { has_variable = unassimilated_refugees } }
			set_variable = { unassimilated_refugees = 0 }
		}
		add_to_variable = { unassimilated_refugees = 1000 }
		on_refugee_change = yes
	}
}

#Diplomatic Parasite, Mythic: a neighbour is goaded. ROOT is the neighbour, FROM the Parasite.
country_event = {
	id = special_rulers.17
	title = special_rulers.17.t
	desc = special_rulers.17.d
	picture = GFX_event_NCR_startup
	is_triggered_only = yes
	fire_only_once = no

	option = {
		name = special_rulers.17.a
		ai_chance = {
			base = 50
			modifier = {
				factor = 0.25
				check_variable = { num_of_factories > FROM.num_of_factories }
			}
		}
		add_dynamic_modifier = { modifier = FOE_goaded_modifier }
		set_variable = { FOE_goaded_by = FROM }
		FROM = { country_event = { id = special_rulers.19 } }
	}

	option = {
		name = special_rulers.17.b
		ai_chance = {
			base = 50
			modifier = {
				factor = 3
				check_variable = { num_of_factories > FROM.num_of_factories }
			}
			modifier = {
				factor = 0.25
				check_variable = { num_of_factories < FROM.num_of_factories }
			}
		}
		declare_war_on = {
			target = FROM
			type = annex_everything
		}
		FROM = {
			add_dynamic_modifier = { modifier = FOE_provoked_modifier days = 100 }
			country_event = { id = special_rulers.18 }
		}
	}
}
