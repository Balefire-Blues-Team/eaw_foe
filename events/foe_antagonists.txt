#DEBUG menu
country_event = {
	id = foe_antagonists.2
	title = foe_antagonists.2.t
	desc = foe_antagonists.2.d
	picture = GFX_event_test
	is_triggered_only = yes

	# Reroll the Antagonists
	option = {
		name = foe_antagonists.2.a
		set_global_flag = FOE_campaign_antagonists_enabled
		FOE_clear_antagonists = yes
		FOE_select_antagonists = yes
	}
	# Toggle Brutal
	option = {
		name = foe_antagonists.2.b
		if = {
			limit = { has_global_flag = FOE_campaign_antagonists_brutal }
			clr_global_flag = FOE_campaign_antagonists_brutal
		}
		else = {
			set_global_flag = FOE_campaign_antagonists_enabled
			set_global_flag = FOE_campaign_antagonists_brutal
		}
		FOE_log_antagonists = yes
	}
	option = {
		name = foe_antagonists.2.c
		FOE_log_antagonists = yes
	}
	# Switch the system off
	option = {
		name = foe_antagonists.2.e
		FOE_clear_antagonists = yes
		clr_global_flag = FOE_campaign_antagonists_enabled
		clr_global_flag = FOE_campaign_antagonists_brutal
	}
	# Run the monthly update now (rubber band + expansion), in each Antagonist's own scope.
	option = {
		name = foe_antagonists.2.g
		every_country = {
			limit = { has_country_flag = FOE_antagonist }
			country_event = { id = foe_antagonists.3 }
		}
	}
	# Run both timers out now
	option = {
		name = foe_antagonists.2.h
		every_country = {
			limit = { has_country_flag = FOE_antagonist }
			if = {
				limit = { has_global_flag = FOE_campaign_antagonists_brutal }
				set_variable = { FOE_antagonist_stall_months = 2 }
				set_variable = { FOE_antagonist_nudge_months = 0 }
			}
			else = {
				set_variable = { FOE_antagonist_stall_months = 5 }
				set_variable = { FOE_antagonist_nudge_months = 1 }
			}
			country_event = { id = foe_antagonists.3 }
		}
	}
	# Force Contact for every Antagonist
	option = {
		name = foe_antagonists.2.i
		every_country = {
			limit = {
				has_country_flag = FOE_antagonist
				NOT = { check_variable = { FOE_antagonist_phase > 0 } }
			}
			country_event = { id = foe_antagonists.6 }
		}
	}
	# Start the Rival's strike countdown
	option = {
		name = foe_antagonists.2.j
		if = {
			limit = { has_variable = global.FOE_antagonist_rival }
			var:global.FOE_antagonist_rival = {
				country_event = { id = foe_antagonists.7 }
			}
		}
	}
	# The Rival is annexed by an AI neighbour (tests an AI kill)
	option = {
		name = foe_antagonists.2.k
		if = {
			limit = {
				has_variable = global.FOE_antagonist_rival
				var:global.FOE_antagonist_rival = { exists = yes }
			}
			var:global.FOE_antagonist_rival = {
				random_neighbor_country = {
					limit = {
						is_ai = yes
						NOT = { has_country_flag = FOE_antagonist }
					}
					annex_country = {
						target = PREV
						transfer_troops = no
					}
				}
			}
		}
	}
	# Run the refill timers out now
	option = {
		name = foe_antagonists.2.l
		if = {
			limit = { has_global_flag = FOE_antagonist_fallen_slot_1 }
			country_event = { id = foe_antagonists.10 }
		}
		if = {
			limit = { has_global_flag = FOE_antagonist_fallen_slot_2 }
			country_event = { id = foe_antagonists.11 }
		}
		if = {
			limit = { has_global_flag = FOE_antagonist_fallen_slot_3 }
			country_event = { id = foe_antagonists.12 }
		}
	}
	# End every running strike countdown now: the wars start
	option = {
		name = foe_antagonists.2.m
		every_country = {
			limit = { has_country_flag = FOE_antagonist_strike_pending }
			country_event = { id = foe_antagonists.14 }
		}
	}
	option = {
		name = foe_antagonists.2.f
	}
}

# Hidden. A day after an Antagonist falls: civil-war successor, defeated by the player, or refill timer
country_event = {
	id = foe_antagonists.8
	hidden = yes
	is_triggered_only = yes

	immediate = {
		FOE_update_antagonist_player = yes
		# A copy queued on a former player country does nothing; the player has its own
		if = {
			limit = { tag = var:global.FOE_antagonist_player }
			FOE_resolve_antagonist_falls = yes
		}
	}
}

# Hidden. Debug: ends this Antagonist's strike countdown now and declares the war (FROM = the player)
country_event = {
	id = foe_antagonists.14
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { has_country_flag = FOE_antagonist_strike_pending }
			if = {
				limit = { has_global_flag = FOE_campaign_antagonists_brutal }
				remove_targeted_decision = { target = FROM decision = FOE_antagonist_strike_mission_brutal }
			}
			else = {
				remove_targeted_decision = { target = FROM decision = FOE_antagonist_strike_mission }
			}
			clr_country_flag = FOE_antagonist_strike_pending
			FOE_antagonist_first_strike = yes
		}
	}
}
