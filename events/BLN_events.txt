add_namespace = blankevent

## Skyline goes to BLN after PAR is no more.
country_event = {
	immediate = {log = "[GetDateText]: [Root.GetName]: event blankevent.1"}
	id = blankevent.1
	title = blankevent.1.t
	desc = blankevent.1.d
	fire_only_once = yes
	trigger = {
		original_tag = BLN
		NOT = { country_exists = PAR }
	}
	option = {
		name = blankevent.1.a
		ai_chance = { factor = 1 }
		hidden_effect = {
			PAR_char_Skyline = {
				set_nationality = BLN
			}
			remove_advisor_role = { character = PAR_char_Skyline slot = army_chief }
			add_advisor_role = {
				character = PAR_char_Skyline
				advisor = {
					slot = army_chief
					idea_token = PAR_Skyline_exiled_adv
					cost = 150
					allowed = {
						original_tag = BLN
					}
					visible = {
						original_tag = BLN
					}
					available = {
						original_tag = BLN
						NOT = {
							country_exists = PAR
						}
					}
					traits = {
						skyline_exiled_trait
					}
					ai_will_do = {
						factor = 1
					}
				}
			}
		}
	}
}
